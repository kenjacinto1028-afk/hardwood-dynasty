def ok(n, c, x=''): res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else ''))
CLR = "document.querySelector('#modal').innerHTML=''"
async def ev(pg, js): return await pg.evaluate(js)
async def go(pg, h):

async def tc(pg, sel='#main'): return await ev(pg, f"(document.querySelector('{sel}')||{{textContent:''}}).textContent")
def clean(t): return '{' not in t and 'undefined' not in t and 'NaN' not in t
async def player(pg, pid):
    await ev(pg, CLR); await ev(pg, f"window.__pbl.ui.player={pid};window.__pbl.render()"); await pg.wait_for_timeout(500)
async def newgame(b, w, h, mobile):
    ctx = await b.new_context(viewport={'width': w, 'height': h}, is_mobile=mobile, has_touch=mobile)
    pg = await ctx.new_page(); errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
    await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
    await pg.click('[data-setmode="team"]'); await pg.click('[data-setgames="58"]'); await pg.click('#create'); await pg.wait_for_timeout(4500); await ev(pg, CLR)
    return pg, errs
