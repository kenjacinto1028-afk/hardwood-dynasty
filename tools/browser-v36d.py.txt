# v36d B: season to-do list rules vs game state, on the franchise save (1300 + 390px). Base: browser-v30t (new league) + browser-v34 (save loader).
# gunzip -c tools/snap/nba-team-s10.json.gz > tools/_t1.json; rename to .py, serve the game folder (python3 -m http.server PORT), then: python3 browser-v36d.py PORT [PAGE] [SAVE]
import asyncio, sys, json
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8796'
PAGE = sys.argv[2] if len(sys.argv) > 2 else 'index.html'
SAVE = sys.argv[3] if len(sys.argv) > 3 else 'tools/_t1.json'
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui,todo=window.__pbl.v30.todo;return eval(" + json.dumps(js) + ")})()")
async def render(pg, js='0', wait=500):
    await ev(pg, "(()=>{" + js + ";window.__pbl.render()})()"); await pg.wait_for_timeout(wait)
async def st(pg, id): return await ev(pg, f"(todo.list().find(x=>x.id==='{id}')||{{}}).done")
# independent copies of what each auto rule claims (league globals only)
EXP = {
 's1c': "watchList(L).filter(p=>p.prospect).length>=3",
 's1e': "!!todo.state().seen.train||rosterOf(L,L.teams[L.user]).filter(p=>!p.prospect&&p.age<=25).every(p=>!!p.focus)",
 's1g': "rosterOf(L,L.teams[L.user]).some(p=>p.core)",
 's1i': "!userRosterProblem(L)&&twOf(L,L.teams[L.user]).length>=TW_MAX",
 's3f': "L.phase!=='regular'||L.day>=deadlineDay(L)",
}
UNH = "(ps=>ps.some(p=>(p.morale!=null&&p.morale<40)||p.tradeReq===L.season))"
async def run(width):
    async with async_playwright() as p:
        b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
        pg = await (await b.new_context(viewport={'width': width, 'height': 900})).new_page(); errs = []
        pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/{PAGE}'); await pg.wait_for_timeout(2000)
        await pg.evaluate("""async (f) => { const txt = await (await fetch(f)).text();
          const db = await new Promise((res, rej) => { const r = indexedDB.open('pbl-league', 1); r.onupgradeneeded = () => r.result.createObjectStore('s'); r.onsuccess = () => res(r.result); r.onerror = () => rej(r.error); });
          const put = (k, v) => new Promise((res, rej) => { const tx = db.transaction('s', 'readwrite'); tx.objectStore('s').put(v, k); tx.oncomplete = res; tx.onerror = () => rej(tx.error); });
          await put('league:slot-t36', txt); await put('slots', JSON.stringify([{ id: 'slot-t36', name: 't36', mode: 'team', season: 11, phase: 'regular', updated: Date.now() }])); await put('current', 'slot-t36'); }""", SAVE)
        await pg.reload(); await pg.wait_for_timeout(6000)
        W = f'{width}px: '
        clr = "document.querySelector('#modal')&&(document.querySelector('#modal').innerHTML='');ui.player=null;ui.game=null;ui.team=null;ui.coach=null"
        await ev(pg, clr)
        ok(W + 'franchise save loaded, to-do on', await ev(pg, "L.user!=null&&L.mode==='team'&&todo.on()"), await ev(pg, "[L.phase,L.day,L.season]"))
        ok(W + 'old save: L.todo missing loads, state made with stop off', await ev(pg, "(()=>{delete L.todo;const T=todo.state();return T.stop===false&&T.s===L.season})()"))
        await render(pg, clr + ";history.replaceState(null,'','#myteam');ui.view='myteam';ui.teamTab='overview';ui.mtab='roster'", 900)
        ok(W + 'panel on My team', await ev(pg, "!!document.querySelector('#todo')"))
        ok(W + 'header chip shown', await ev(pg, "(()=>{const c=document.getElementById('todochip');return !!c&&!c.hidden})()"))
        # rules vs state
        for k, js in EXP.items():
            a, e = await st(pg, k), await ev(pg, js)
            ok(W + f'{k} matches game state', a == e, (a, e))
        # lineup: only from the Roster tab
        await render(pg, "ui.twin=Object.assign({},ui.twin,{rot:'Lineup & rotation'});ui.mtab='cap';todo.state().seen.lineup=0")
        ok(W + 's1f stays open when Lineup was picked earlier but the Salary cap tab shows', await st(pg, 's1f') is False)
        await render(pg, "ui.mtab='roster'")
        ok(W + 's1f ticks on the Roster tab with Lineup & rotation open', await st(pg, 's1f') is True)
        # locker room: harmony
        await render(pg, "L.dramaOpen=null;L.mtg=null;rosterOf(L,L.teams[L.user]).forEach(p=>{p.morale=Math.max(p.morale||55,60);p.tradeReq=null});lrOf(L.teams[L.user]).v=30")
        ok(W + 's1j open with harmony in the red (30)', await st(pg, 's1j') is False)
        await render(pg, "lrOf(L.teams[L.user]).v=60")
        ok(W + 's1j ticks: harmony 60, no drama / meeting / unhappy', await st(pg, 's1j') is True)
        await render(pg, "rosterOf(L,L.teams[L.user])[5].morale=30")
        ok(W + 's1j open with an unhappy player', await st(pg, 's1j') is False)
        ok(W + 's6h follows top-5 unhappy only', await st(pg, 's6h') == await ev(pg, "!userRosterProblem(L)&&!dramaOpen(L)&&!" + UNH + "(rotationOf(L,L.teams[L.user]).slice(0,5))"))
        await render(pg, "rosterOf(L,L.teams[L.user])[5].morale=60")
        # promises: a game-count promise from last season is not 'at risk' now
        await render(pg, "L.promises=(L.promises||[]).filter(x=>!x.t36);const p=rosterOf(L,L.teams[L.user])[3],t=L.teams[L.user];L.promises.push({t36:1,id:9e6,pid:p.id,tid:L.user,k:'min',s:L.season-1,st:'open',due:{s:L.season-1,g:(t.w+t.l)+3}})")
        ok(W + 's2b not flagged by last season\'s game-count promise', await st(pg, 's2b') is True)
        await render(pg, "const pr=L.promises.find(x=>x.t36);pr.s=L.season;pr.due.s=L.season")
        ok(W + 's2b open with a promise due in 3 games this season', await st(pg, 's2b') is False)
        await render(pg, "L.promises=L.promises.filter(x=>!x.t36)")
        # extensions rule: open iff a willing extension candidate exists (canExtend)
        ok(W + 's1h done when no player can extend', (await st(pg, 's1h')) is True or await ev(pg, "rosterOf(L,L.teams[L.user]).some(p=>canExtend(L,p))"))
        # deadline interview row only shows while open
        ok(W + 's3e hidden when no interview is open', await ev(pg, "!todo.list().some(x=>x.id==='s3e')") == (not await ev(pg, "!!ivOpen(L)")))
        # sim stop
        r = await ev(pg, "(()=>{const T=todo.state();T.stop=true;T.stopped={};const dl=deadlineDay(L);const save=L.day;L.day=Math.max(1,dl-10);ui.mtab='roster';const a=todo.stopEnd(dl+5),star=todo.list().filter(x=>x.star&&!x.done&&!x.hidden&&['start','early','mid'].includes(x.g)).length;T.stop=false;const b=todo.stopEnd(dl+5);L.day=save;return [a,dl-3,star,b,dl+5]})()")
        ok(W + 'sim stop: 3 days before the deadline when a star item is open, off = no stop', (r[0] == r[1] if r[2] else r[0] == r[4]) and r[3] == r[4], r)
        # nothing in commissioner / jobless / autopilot
        await render(pg, "L.mode='commish'"); ok(W + 'commissioner: no panel, no chip', await ev(pg, "!todo.on()&&!document.querySelector('#todo')&&(!document.getElementById('todochip')||document.getElementById('todochip').hidden)"))
        await render(pg, "L.mode='team';L.jobless=true"); ok(W + 'jobless: off', await ev(pg, "!todo.on()"))
        await render(pg, "L.jobless=false;ui.auto=true"); ok(W + 'autopilot: off', await ev(pg, "!todo.on()"))
        await render(pg, "ui.auto=false;ui.mtab='roster'", 800)
        ok(W + 'back on: panel shown', await ev(pg, "todo.on()&&!!document.querySelector('#todo')"))
        ok(W + 'My team: no sideways scroll', await ev(pg, "document.documentElement.scrollWidth<=window.innerWidth+2"))
        await pg.screenshot(path=f'/tmp/v36d-todo-{width}.png')
        ok(W + 'no page errors', not errs, errs[:3])
        await b.close()
async def main():
    for w in (1300, 390): await run(w)
    print('\n'.join(res)); print(f"{sum(r.startswith('PASS') for r in res)}/{len(res)} PASS")
asyncio.run(main())
