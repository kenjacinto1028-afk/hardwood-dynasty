# v31 UI check: a player overseas (p.abroad): player page label + no negotiate, League > Players team column + FA / Overseas filters. PBL 16, desktop 1300px + 390px.
# Rename to .py, serve the game folder (python3 -m http.server PORT), then: python3 browser-v31.py PORT
import asyncio, sys, json
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8791'
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
CLR = "document.querySelector('#modal').innerHTML=''"
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui;return eval(" + json.dumps(js) + ")})()")
async def tc(pg, sel='#main'): return await ev(pg, f"(document.querySelector('{sel}')||{{textContent:''}}).textContent")
async def noscroll(pg): return await ev(pg, "document.documentElement.scrollWidth <= window.innerWidth + 1")
clean = lambda t: '{' not in t and 'undefined' not in t and 'NaN' not in t
async def main():
    async with async_playwright() as p:
        b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
        ctx = await b.new_context(viewport={'width': 1300, 'height': 900}); pg = await ctx.new_page(); errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
        await pg.click('[data-setmode="team"]'); await pg.click('[data-setnba="0"]'); await pg.click('[data-setsize="16"]')
        await pg.click('[data-setgames="58"]'); await pg.click('#create'); await pg.wait_for_timeout(4500); await ev(pg, CLR)
        # stage one abroad player the way sendAbroad + pipeBuild leave him
        info = json.loads(await ev(pg, """(()=>{const p=L.fa.map(id=>L.players[id]).filter(q=>q&&!q.prospect).sort((a,b)=>b.ovr-a.ovr)[0];
          sendAbroad(L,p);p.club={lg:'EABL',t:0,s:L.season};p.tx.push({s:L.season,t:'Went overseas: signed with '+pipeTeamName('EABL',0)+' (EABL)'});
          return JSON.stringify({id:p.id,name:p.name,lab:abroadLabel(p),ovr:p.ovr,inFA:L.fa.includes(p.id)});})()"""))
        ok('staged player is off L.fa, label has club name', not info['inFA'] and info['lab'] == 'Overseas (London Monarchs)', info['lab'])
        pid, nm, lab = info['id'], info['name'], info['lab']
        async def page_checks(tag):
            await ev(pg, CLR); await ev(pg, f"ui.negOpen=null;ui.player={pid};window.__pbl.render()"); await pg.wait_for_timeout(600)
            t = await tc(pg)
            ok(f'{tag} player page shows "{lab}"', lab in t, t[:120].replace('\n', ' '))
            ok(f'{tag} player page: not shown as Free agent, no Negotiate button', 'Free agent' not in t.split('Contract')[0] and await ev(pg, "!document.querySelector('[data-negotiate],#negbar')"), t[:80])
            ok(f'{tag} player page: transaction line present, clean text', 'Went overseas' in t and clean(t))
            ok(f'{tag} player page: no sideways scroll', await noscroll(pg), await ev(pg, "document.documentElement.scrollWidth"))
        await page_checks('[1300]')
        await pg.set_viewport_size({'width': 390, 'height': 844}); await pg.wait_for_timeout(400)
        await page_checks('[390]')
        await pg.set_viewport_size({'width': 1300, 'height': 900}); await pg.wait_for_timeout(300)
        # League > Players: Stats tab (vPlayers, ui.pfTeam) and the Info directory (vPlayerDir, ui.pdTeam), rendered through their routes
        async def view(setup, hsh):
            await ev(pg, CLR); await ev(pg, "ui.player=null;ui.game=null;ui.coach=null"); await ev(pg, "location.hash='#standings'"); await pg.wait_for_timeout(250)
            await ev(pg, setup); await ev(pg, f"location.hash='#{hsh}'"); await pg.wait_for_timeout(900); await ev(pg, "window.__pbl.render()"); await pg.wait_for_timeout(500)
            return await ev(pg, "document.querySelector('#main').innerHTML")
        base = "ui.pq='';ui.pfAgeLo=ui.pfAgeHi='';ui.pfPos='all';ui.pAll=1;ui.pfType='';ui.statTab='players';ui.pdq='';ui.pdPos='all';ui.pdAgeLo=ui.pdAgeHi=ui.pdMin='';ui.pdNat='all';ui.pdTab='info';"
        h = {}
        for k in ['all', 'fa', 'ov']:
            h['pl_' + k] = await view(base + f"ui.pfTeam='{k}'", 'stats')
            h['dir_' + k] = await view(base + f"ui.pdTeam='{k}'", 'players')
        ok('League > Players (Stats): "All teams" shows him with the Overseas label', nm in h['pl_all'] and lab in h['pl_all'])
        ok('League > Players (Stats): Free agents filter excludes him', nm not in h['pl_fa'])
        ok('League > Players (Stats): Overseas filter lists him, with the label', nm in h['pl_ov'] and lab in h['pl_ov'])
        ok('League > Players (Info): "All teams" shows him as Overseas, FA filter excludes him, Overseas filter lists him', lab in h['dir_all'] and nm not in h['dir_fa'] and nm in h['dir_ov'] and lab in h['dir_ov'])
        ok('League > Players: both filters offer an Overseas option', 'value="ov"' in h['pl_all'] and 'value="ov"' in h['dir_all'])
        for w, hh, tag in [(1300, 900, '[1300]'), (390, 844, '[390]')]:
            await pg.set_viewport_size({'width': w, 'height': hh}); await pg.wait_for_timeout(300)
            for nme, setup, hsh in [('Info', base + "ui.pdTeam='ov'", 'players'), ('Stats', base + "ui.pfTeam='ov'", 'stats')]:
                html = await view(setup, hsh); t = await tc(pg)
                ok(f'{tag} League > Players {nme} with Overseas filter: clean text, lists him', clean(t) and nm in t, t[:60].replace('\n', ' '))
                ok(f'{tag} League > Players {nme}: no sideways scroll', await noscroll(pg), await ev(pg, "document.documentElement.scrollWidth"))
        ok('no page errors', not errs, errs[:2])
        await b.close()
    bad = [r for r in res if r.startswith('FAIL')]
    print(f'{len(res) - len(bad)}/{len(res)} pass'); print('\n'.join(bad)); sys.exit(1 if bad else 0)
asyncio.run(main())
