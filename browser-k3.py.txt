# v27.4 K3 checks (coach sliders UI). Serve the folder, then: python3 browser-k3.py PORT
import asyncio, sys
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8782'
res = []
def ok(n, c, x=''): res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else ''))
async def main():
    async with async_playwright() as p:
        b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
        pg = await (await b.new_context(viewport={'width': 1300, 'height': 900})).new_page()
        errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
        await pg.click('[data-setmode="team"]'); await pg.click('[data-setnba="1"]'); await pg.wait_for_timeout(300); await pg.click('#create'); await pg.wait_for_timeout(4500)
        await pg.evaluate("document.querySelector('#modal').innerHTML=''")
        txt = "document.querySelector('#main').innerText"
        # own coach page
        await pg.evaluate("(()=>{const L=window.__pbl.L;const c=L.teams[L.user].coach;window.__pbl.ui.coach=c.id;window.__pbl.render()})()"); await pg.wait_for_timeout(500)
        t = await pg.evaluate(txt)
        ok('S1 Philosophy panel on own coach', 'PHILOSOPHY' in t.upper() and 'Reset to system default' in t)
        n = await pg.evaluate("document.querySelectorAll('input[data-sl]').length"); dis = await pg.evaluate("document.querySelectorAll('input[data-sl]:disabled').length")
        ok('S2 8 sliders, all editable', n == 8 and dis == 0, f'{n}/{dis}')
        ok('S2 trade-off line shown', 'Few threes' in t or 'Walk it up' in t)
        # move pace to 100
        before = await pg.evaluate("(()=>{const L=window.__pbl.L;const c=L.teams[L.user].coach;return [c.pace,c.depth]})()")
        await pg.evaluate("(()=>{const el=document.querySelector('input[data-sl=pace]');el.value=100;el.dispatchEvent(new Event('input',{bubbles:true}));el.dispatchEvent(new Event('change',{bubbles:true}))})()"); await pg.wait_for_timeout(600)
        await pg.evaluate("(()=>{const el=document.querySelector('input[data-sl=depth]');el.value=100;el.dispatchEvent(new Event('change',{bubbles:true}))})()"); await pg.wait_for_timeout(600)
        after = await pg.evaluate("(()=>{const L=window.__pbl.L;const c=L.teams[L.user].coach;return [c.pace,c.depth,c.sl&&c.sl.pace,c.sl&&c.sl.depth]})()")
        ok('S4 pace/depth change', after[0] > before[0] and after[1] >= before[1] and after[2] == 100 and after[3] == 100, f'{before} -> {after}')
        await pg.evaluate("window.__pbl.save()"); await pg.wait_for_timeout(1500)
        await pg.reload(); await pg.wait_for_timeout(3500)
        kept = await pg.evaluate("(()=>{const L=window.__pbl.L;if(!L)return null;const c=L.teams[L.user].coach;return [c.pace,c.sl&&c.sl.pace]})()")
        ok('S4 reload keeps it', kept is not None and kept[1] == 100, str(kept))
        # reset
        await pg.evaluate("(()=>{const L=window.__pbl.L;window.__pbl.ui.coach=L.teams[L.user].coach.id;window.__pbl.render()})()"); await pg.wait_for_timeout(500)
        await pg.click('#sl-reset'); await pg.wait_for_timeout(600)
        rs = await pg.evaluate("(()=>{const L=window.__pbl.L;const c=L.teams[L.user].coach;const d=slDefault(c),s=slOf(c);return SL_KEYS.every(k=>s[k]===d[k])})()")
        ok('S2 reset to default', rs)
        # AI coach read only
        await pg.evaluate("(()=>{const L=window.__pbl.L;const t=L.teams.find((x,i)=>i!==L.user);window.__pbl.ui.coach=t.coach.id;window.__pbl.render()})()"); await pg.wait_for_timeout(500)
        dis = await pg.evaluate("[document.querySelectorAll('input[data-sl]').length,document.querySelectorAll('input[data-sl]:disabled').length,!!document.querySelector('#sl-reset')]")
        ok('S1 AI coach read-only', dis[0] == 8 and dis[1] == 8 and not dis[2], str(dis))
        # S3 market chips
        await pg.evaluate("(()=>{const U=window.__pbl.ui;U.coach=null;U.view='coaches';U.stab='coaches';window.__pbl.render()})()"); await pg.wait_for_timeout(600)
        chips = await pg.evaluate("document.querySelectorAll('.sl-chip').length")
        ok('S3 coach chips in market', chips > 0, str(chips))
        t = await pg.evaluate(txt)
        ok('no raw template text', '${' not in t and 'undefined' not in t and 'NaN' not in t)
        ok('no page errors', not errs, str(errs[:2]))
        await b.close()
    print('\n'.join(res))
asyncio.run(main())
