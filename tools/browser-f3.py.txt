# F1-F3 browser check: serve game folder (python3 -m http.server PORT), run python3 browser-f3.py PORT
import asyncio, sys, json
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8791'
CLR = "document.querySelector('#modal').innerHTML=''"
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L;return eval(" + json.dumps(js) + ")})()")
def ok(n, c, x=''): print(('PASS ' if c else 'FAIL ') + n + (f' [{x}]' if x else ''), flush=True)
async def main():
  async with async_playwright() as p:
    b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
    pg = await (await b.new_context(viewport={'width': 1300, 'height': 900})).new_page(); errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
    await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
    await pg.click('[data-setnba="1"]'); await pg.wait_for_timeout(300); await pg.click('[data-setmode="team"]')
    if await ev(pg, "!!document.querySelector('[data-setgames=\"58\"]')"): await pg.click('[data-setgames="58"]')
    await pg.click('#create'); await pg.wait_for_timeout(4500); await ev(pg, CLR)
    # F1 text check on main tabs
    for h in ['standings', 'teams', 'awards']:
        await ev(pg, f"location.hash='#{h}'"); await pg.wait_for_timeout(600)
        t = await ev(pg, "document.querySelector('#main').innerText")
        ok(f'F1 {h}: no North/South', 'North' not in t and 'South' not in t and 'East' in t or h == 'awards', [w for w in ('North', 'South') if w in t])
    # F3
    await ev(pg, "location.hash='#pipeline'"); await pg.wait_for_timeout(800)
    await ev(pg, "(()=>{const s=document.querySelector('select[data-pipesel]');s.value='-1';s.dispatchEvent(new Event('change',{bubbles:true}))})()"); await pg.wait_for_timeout(600)
    rows = await ev(pg, "document.querySelectorAll('table[data-sid=\"pipe-all\"] tbody tr').length")
    ok('F3 all players table has rows', rows > 20, rows)
    async def setf(k, v):
        await ev(pg, f"(()=>{{const e=document.querySelector('[data-pipef=\"{k}\"]');e.value='{v}';e.dispatchEvent(new Event('change',{{bubbles:true}}))}})()"); await pg.wait_for_timeout(500)
    await setf('ovr', 50); r1 = await ev(pg, "document.querySelectorAll('table[data-sid=\"pipe-all\"] tbody tr').length"); ok('F3 min OVR narrows', r1 < rows, r1)
    await setf('pos', 'C'); r2 = await ev(pg, "[...document.querySelectorAll('table[data-sid=\"pipe-all\"] tbody tr td:nth-child(3)')].map(x=>x.textContent)")
    ok('F3 position filter', len(r2) > 0 and all('C' in x for x in r2), len(r2))
    await setf('amin', 19); await setf('amax', 20); r3 = await ev(pg, "[...document.querySelectorAll('table[data-sid=\"pipe-all\"] tbody tr td:nth-child(4)')].map(x=>+x.textContent)")
    ok('F3 age range', all(19 <= a <= 20 for a in r3), len(r3))
    await setf('pot', 60); ok('F3 min POT runs', await ev(pg, "!!document.querySelector('table[data-sid=\"pipe-all\"]')"))
    await pg.click('[data-pipef-clear]'); await pg.wait_for_timeout(500)
    ok('F3 clear restores', await ev(pg, "document.querySelectorAll('table[data-sid=\"pipe-all\"] tbody tr').length") == rows)
    ok('no page errors', not errs, errs[:2])
    await b.close()
asyncio.run(main())
