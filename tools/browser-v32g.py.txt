# v32g browser check (1300 + 390px): mid-level warnings in the offer popup (real cap exception), saved rotation chart with an
# injured player (game page note, editor note, his squares can be cleared). Rename to .py, serve the game folder, python3 browser-v32g.py PORT
import asyncio, sys, json
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8794'
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui;return eval(" + json.dumps(js) + ")})()")
async def render(pg, js='0', wait=500):
    await ev(pg, "(()=>{" + js + ";window.__pbl.render()})()"); await pg.wait_for_timeout(wait)
async def run(width):
    async with async_playwright() as p:
        b = await p.chromium.launch(); pg = await (await b.new_context(viewport={'width': width, 'height': 900})).new_page(); errs = []
        pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
        await pg.click('[data-setmode="team"]'); await pg.click('[data-setnba="1"]'); await pg.click('[data-setgames="58"]'); await pg.click('#create'); await pg.wait_for_timeout(4500)
        await ev(pg, "document.querySelector('#modal').innerHTML=''")
        W = f'{width}px: '
        noscroll = lambda: ev(pg, "document.documentElement.scrollWidth<=window.innerWidth+2")
        await ev(pg, "(()=>{L.phase='regular';for(let d=0;d<3;d++)simDay(L)})()")
        # ---- offer popup: payroll just over the cap, under the tax line ----
        await ev(pg, "(()=>{const t=L.teams[L.user];t.mleUsed=null;t.dead=[];const ps=t.roster.map(id=>L.players[id]).sort((a,b)=>b.contract.amt-a.contract.amt);for(const p of ps){if(payroll(L,t)<=CAP-3)break;p.contract.amt=MINSAL}t.dead=[{pid:-9,name:'x',amt:Math.max(0,CAP+2-payroll(L,t)),exp:capSeason(L)}];while(t.roster.length>13)release(L,rosterOf(L,t).sort((a,b)=>a.ovr-b.ovr)[0]);ui.neg={}})()")
        fa = await ev(pg, "(()=>{const p=L.fa.map(id=>L.players[id]).filter(p=>p&&!p.prospect&&!p.retired).sort((a,b)=>b.ovr-a.ovr)[0];return p.id})()")
        await render(pg, f"ui.view=null;ui.game=null;ui.player={fa}", 500)
        await pg.click('#negbar [data-nego]'); await pg.wait_for_timeout(700)
        await ev(pg, "(()=>{const i=document.querySelector('#neg-amt');i.value='8';i.dispatchEvent(new Event('change',{bubbles:true}))})()"); await pg.wait_for_timeout(700)
        el = await ev(pg, "(()=>{const c=document.querySelector('#neg-mle');return c?[c.disabled,c.checked]:null})()")
        mb = await ev(pg, "(()=>{const m=document.querySelector('#mlebox');return m?m.textContent:''})()")
        ok(W + 'over the cap: Use mid-level enabled + ticked, no "needs" warning while ticked', el == [False, True] and 'needs the mid-level' not in mb, f'{el} {mb[:120]}')
        await ev(pg, "(()=>{document.querySelector('#neg-mle').click()})()"); await pg.wait_for_timeout(700)
        mb2 = await ev(pg, "(()=>{const m=document.querySelector('#mlebox');return m?m.textContent:''})()")
        ok(W + 'unticked over the cap: warning that the offer needs the mid-level', 'needs the mid-level' in mb2, mb2[:200])
        await ev(pg, "(()=>{const i=document.querySelector('#neg-amt');i.value='40';i.dispatchEvent(new Event('change',{bubbles:true}))})()"); await pg.wait_for_timeout(700)
        mb3 = await ev(pg, "(()=>{const m=document.querySelector('#mlebox');return m?m.textContent:''})()")
        ok(W + 'offer over the MLE amount: only a minimum deal fits', 'Only a minimum deal fits' in mb3, mb3[:220])
        ok(W + 'offer popup: no horizontal scroll', await noscroll())
        await ev(pg, "document.querySelector('#modal').innerHTML='';ui.negOpen=null")
        # ---- saved chart with an injured player ----
        await ev(pg, "(()=>{const t=L.teams[L.user];for(const id of t.roster){L.players[id].injury=0;L.players[id].injName=null}const ps=t.roster.map(id=>L.players[id]).sort((a,b)=>b.ovr-a.ovr).slice(0,9),a=ps.slice(0,5).map(p=>p.id),b=[ps[5],ps[6],ps[7],ps[8],ps[0]].map(p=>p.id);t.rotChart={q:[0,1,2,3].map(q=>[a,a,b,a])};const o=L.players[b[1]];o.injury=6;o.injName='Ankle sprain';window.__outId=o.id;let g0=L.games.length;for(let d=0;d<6;d++){simDay(L);if(L.games.slice(g0).some(x=>x.home===L.user||x.away===L.user))break;}})()")
        gid = await ev(pg, "(()=>{const g=L.games.filter(x=>x.tl&&(x.home===L.user||x.away===L.user)).slice(-1)[0];return g?g.id:null})()")
        await render(pg, f"ui.view=null;ui.player=null;ui.game={gid};ui.liveMode=null", 700)
        note = await ev(pg, "(()=>{const x=[...document.querySelectorAll('.rotchart p.muted')].map(e=>e.textContent).join(' | ');return x})()")
        ok(W + 'game page chart: source = your chart, note names the stand-in', 'Your saved rotation chart' in note and 'Chart changed for this game' in note and 'took his' in note, note[:260])
        ok(W + 'game page: no horizontal scroll', await noscroll())
        await render(pg, "ui.game=null;ui.liveMode=null;ui.player=null;ui.view='myteam';ui.mtab='roster';ui.teamTab='overview';ui.twin=Object.assign({},ui.twin,{rot:'Rotation chart'});ui.rcDraft=null", 700)
        en = await ev(pg, "(()=>{const id=window.__outId;const on=[...document.querySelectorAll('[data-rcb^=\"'+id+'|\"]')];return [on.length,on.filter(b=>b.classList.contains('on')&&!b.disabled).length,on.filter(b=>!b.classList.contains('on')&&b.disabled).length]})()")
        tx = await ev(pg, "(()=>{return [...document.querySelectorAll('.msg')].map(e=>e.textContent).join(' | ')})()")
        ok(W + 'editor: injured player squares can be cleared (on = enabled), empty ones locked', en and en[0] == 16 and en[1] == 4 and en[2] == 12, json.dumps(en))
        ok(W + 'editor: note says a stand-in takes his blocks; last game note', 'takes his blocks' in tx and 'stand-in' in tx, tx[:300])
        ok(W + 'editor: save not blocked by the injury', await ev(pg, "(()=>{const b=document.querySelector('[data-rcact=\"save\"]');return !!b})()"))
        ok(W + 'editor: no horizontal scroll', await noscroll())
        ok(W + 'no page errors', not errs, '; '.join(errs)[:200])
        await b.close()
async def main():
    for w in (1300, 390): await run(w)
    print('\n'.join(res)); print(f"\nbrowser-v32g: {sum(r.startswith('PASS') for r in res)}/{len(res)}")
asyncio.run(main())
