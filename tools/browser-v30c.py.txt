# v30c UI checks (Sonnet): #1 Roster moves overview (My team / League), #2 Negotiate popup with live cap numbers, #5 move write-ups,
# #19 Promises Current / Former, #21 Hall of Fame ceremony, #23 interview periods + pools. NBA 30 + PBL 16, desktop 1300px + 390px.
# Rename to .py, serve the game folder (python3 -m http.server PORT), then: python3 browser-v30c.py PORT
import asyncio, sys, json, re
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8789'
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
CLR = "document.querySelector('#modal').innerHTML=''"
def clean(t): return '{' not in t and 'undefined' not in t and 'NaN' not in t
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui;return eval(" + json.dumps(js) + ")})()")
async def tc(pg, sel='#main'): return await ev(pg, f"(document.querySelector('{sel}')||{{textContent:''}}).textContent")
async def go(pg, h):
    await ev(pg, CLR); await ev(pg, "ui.player=null;ui.game=null;ui.coach=null;ui.negOpen=null"); await ev(pg, "location.hash='#standings'"); await pg.wait_for_timeout(200)
    await ev(pg, f"location.hash='#{h}'"); await pg.wait_for_timeout(700)
async def player(pg, pid):
    await ev(pg, CLR); await ev(pg, f"ui.negOpen=null;ui.player={pid};window.__pbl.render()"); await pg.wait_for_timeout(500)
async def noscroll(pg): return await ev(pg, "document.documentElement.scrollWidth <= window.innerWidth + 1")
async def narrow(pg, label, sel='#main'):
    await pg.set_viewport_size({'width': 390, 'height': 844}); await pg.wait_for_timeout(400)
    await ev(pg, "window.__pbl.render()"); await pg.wait_for_timeout(300)
    t = await tc(pg, sel); ok(f'390px: {label}: no sideways scroll', await noscroll(pg), await ev(pg, "document.documentElement.scrollWidth"))
    ok(f'390px: {label}: clean text', clean(t))
    await pg.set_viewport_size({'width': 1300, 'height': 900}); await pg.wait_for_timeout(300)
async def narrow_modal(pg, label):
    await pg.set_viewport_size({'width': 390, 'height': 844}); await pg.wait_for_timeout(400)
    ok(f'390px: {label}: popup fits, no sideways scroll', await noscroll(pg) and await ev(pg, "(()=>{const d=document.querySelector('#modal .dialog');return !d||d.scrollWidth<=d.clientWidth+1})()"), await ev(pg, "document.documentElement.scrollWidth"))
    await pg.set_viewport_size({'width': 1300, 'height': 900}); await pg.wait_for_timeout(300)
WU = r'^[A-Z0-9]{2,4}: [^{}]+\.( Coach fit \d+ > \d+\.)?( Team overall \d+ > \d+\.)?$'

    T = f'[{tag}] '

      const fa=L.fa.map(id=>L.players[id]).filter(p=>p&&!p.retired&&!p.prospect).sort((a,b)=>b.ovr-a.ovr)[0];signContract(L,fa,t.id,MINSAL,1,'signing');

    ok(T + '#1 Roster moves group lands on the Overview and lists Overview / Trades / Free agency / Draft', await ev(pg, "(()=>{const a=[...document.querySelectorAll('#subtabs a')].map(x=>x.dataset.v);return a.join()==='rmoves,trades,fa,draft'&&document.querySelector('#subtabs a.on').dataset.v==='rmoves'})()"))

            t = await tc(pg, '#modal')
            ok(T + '#2 Send offer in the popup gets an answer (accepted dialog, or his reply in the popup)', ('accepted' in t or await ev(pg, "!!document.querySelector('#modal .msg')")) and clean(t), t[:160])

    # ============ #23 draft-week interview (real buttons) ============
    ph = await ev(pg, "L.phase")

    ctx = await b.new_context(viewport={'width': 1300, 'height': 900})

async def main():
    async with async_playwright() as p:
        b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
        await league(b, True, 30, 'NBA')
        await league(b, False, 16, 'PBL16')
        await b.close()
    bad = [r for r in res if r.startswith('FAIL')]
    print(f'{len(res) - len(bad)}/{len(res)} pass'); print('\n'.join(bad))
    sys.exit(1 if bad else 0)
asyncio.run(main())
