import sys, json, statistics as st
for f in sys.argv[1:]:
    rows=[json.loads(l[2:]) for l in open(f) if l.startswith('S ')]
    rows=[r for r in rows if 'err' not in r]
    pay=[r['v32e']['payroll'] for r in rows]
    rs=[r['v32e']['resign'] for r in rows]
    print('==',f,len(rows),'seasons')
    print(' payroll avg', round(st.mean(p['avg'] for p in pay),1), 'per season', [round(p['avg']) for p in pay])
    print(' min team', [round(p['min']) for p in pay], 'underFloor', [p['underFloor'] for p in pay], '<=2:', sum(p['underFloor']<=2 for p in pay))
    print(' maxNext', max(p['max'] for p in pay))
    print(' resign rate', [r['rate'] for r in rs], 'rate60', [r.get('rate60') for r in rs])
    print(' top PER', [r['adv']['per'][0]['per'] for r in rows if r.get('adv')], 'top WS', [r['adv']['ws'][0]['ws'] for r in rows if r.get('adv')])
    print(' ovrTop', [r['adv']['ovrTop'][:3] for r in rows if r.get('adv')])
    y=[]
    for r in rows:
        b=r['v32e']['asks']['bands'].get('56-59|<=25'); y.append(b['max'] if b else None)
    print(' 56-59<=25 max ask', y, 'under60ask10', sum(r['v32e']['asks']['under60ask10N'] for r in rows))
    print(' lgOvr', round(st.mean(r['v32g']['lgOvr'] for r in rows),1))
    # v36a: 60+ re-sign losses by reason (team side t:*, player side p:*), owner cash
    w=[r['v32e'].get('why60') or {} for r in rows]
    tot={}
    for x in w:
        for k,v in x.items():
            if k not in ('n','kept','capHard','capOwner','capIcon','capCleared','capClearFail','ccWhy','ccGap','capClear9','discStay'): tot[k]=tot.get(k,0)+v
    print(' why60 lost', sum(x.get('n',0)-x.get('kept',0) for x in w), 'of', sum(x.get('n',0) for x in w), dict(sorted(tot.items(),key=lambda kv:-kv[1])), 't:team per season', [x.get('t:team',0) for x in w])
    print(' t:cap split hard/owner/icon', sum(x.get('capHard',0) for x in w), sum(x.get('capOwner',0) for x in w), sum(x.get('capIcon',0) for x in w))
    print(' capCleared (kept after moving a bench deal) / clear failed', sum(x.get('capCleared',0) for x in w), sum(x.get('capClearFail',0) for x in w), '| with a 7th-9th man', sum(x.get('capClear9',0) for x in w), '| discount stays (tx)', sum(x.get('discStay',0) for x in w), '| t:cap % of 60+ expiring', round(100*tot.get('t:cap',0)/max(1,sum(x.get('n',0) for x in w)),1))
    cw={}
    for x in w:
        for k,v in (x.get('ccWhy') or {}).items(): cw[k]=cw.get(k,0)+v
    if cw: print(' clear room: why no move', cw, '(noBench = only minimum / protected deals outside the top 9; tooBig = gap over the hard cap > movable bench salary; ownerLimit = payroll after the move past the owner season limit / tax-line rule)')
    c=[r['cash'] for r in rows if r.get('cash')]
    if c: print(' owner cash avg', [round(x['avg']) for x in c], 'min', [round(x['min']) for x in c], 'neg teams', [x['neg'] for x in c])
