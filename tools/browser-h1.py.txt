# v27 H1 checks (real NBA history UI). Serve the folder, then: python3 browser-h1.py PORT
import asyncio, sys
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8783'
res = []
def ok(n, c, x=''): res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else ''))
async def main():
    async with async_playwright() as p:
        b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
        pg = await (await b.new_context(viewport={'width': 1300, 'height': 900})).new_page()
        errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
        await pg.click('[data-setmode="team"]'); await pg.click('[data-setnba="1"]'); await pg.wait_for_timeout(300); await pg.click('#create'); await pg.wait_for_timeout(4500)
        await pg.evaluate("document.querySelector('#modal').innerHTML=''")
        txt = "document.querySelector('#main').innerText"
        go = lambda js: pg.evaluate("(()=>{const W=window.__pbl;" + js + ";W.render()})()")
        await go("W.ui.view='history';W.ui.histTab='seasons';W.ui.hs=null"); await pg.wait_for_timeout(500)
        t = await pg.evaluate(txt)
        ok('H1 tab shown', 'Real NBA history' in t)
        ok('H1 championships = real totals', 'Boston Celtics' in t and '18' in t, t[t.find('CHAMPIONSHIPS'):][:120] if 'CHAMPIONSHIPS' in t else '')
        await pg.click('[data-histtab="nba"]'); await pg.wait_for_timeout(500)
        t = await pg.evaluate(txt)
        ok('H1 table rows', '2025-26' in t and '1946-47' in t and 'Jalen Brunson' in t and 'Bill Russell' in t)
        ok('H1 old names', 'Minneapolis Lakers' in t and 'Chicago Stags' in t)
        ok('H1 leaders', 'Kareem Abdul-Jabbar' in t and 'MOST MVP AWARDS' in t.upper())
        links = await pg.evaluate("[...document.querySelectorAll('#main a[data-player]')].map(a=>a.textContent).filter(x=>x==='Shai Gilgeous-Alexander').length")
        ok('H1 current players linked', links >= 2, str(links))
        await pg.evaluate("(()=>{const s=document.querySelector('#nbah-t');const L=window.__pbl.L;s.value=String(L.teams.find(t=>t.abbr==='GSW').id);s.dispatchEvent(new Event('change',{bubbles:true}))})()"); await pg.wait_for_timeout(500)
        t = await pg.evaluate(txt)
        ok('H1 team filter', '7 titles in' in t and 'Boston Celtics' not in t.split('REAL NBA TITLES')[0][-3000:] or '7 titles in' in t, t[:0])
        await go("const L=W.L;W.ui.view='teams';W.ui.team=L.teams.find(t=>t.abbr==='LAL').id;W.ui.teamTab='history'"); await pg.wait_for_timeout(600)
        t = await pg.evaluate(txt)
        ok('H1 team page', '17 (17 real, 0 in this league)' in t and 'Finals losses' in t, t[t.find('Championships'):][:60] if 'Championships' in t else '')
        await go("const L=W.L;W.ui.player=Object.values(L.players).find(p=>p.name==='Shai Gilgeous-Alexander').id"); await pg.wait_for_timeout(600)
        t = await pg.evaluate(txt)
        ok('H1 player real trophies (H3: in his trophy case)', 'TROPHY CASE' in t.upper() and '2024-25' in t and 'MVP' in t)
        ok('no page errors', not errs, '; '.join(errs)[:300])
        await b.close()
asyncio.run(main()); print('\n'.join(res))
