# v33d browser check (1300 + 390px): "est." on PER / WS seasons worked out on load for a save played before v33c (advBackfill):
# player page Advanced tab (sim seasons tagged, tooltip), season records "(est.)", no page errors, no sideways scroll.
# Save: the pre-v33c saved league from git (git show 3b5d073:tools/snap/nba-commish-s10.json.gz | gunzip > tools/_s1.json).
# Rename to .py, serve the game folder (python3 -m http.server PORT), then: python3 browser-v33d.py PORT
import asyncio, sys, json
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8794'
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui;return eval(" + json.dumps(js) + ")})()")
async def render(pg, js='0', wait=700):
    await ev(pg, "(()=>{" + js + ";window.__pbl.render()})()"); await pg.wait_for_timeout(wait)
async def run(width):
    async with async_playwright() as p:
        b = await p.chromium.launch()
        pg = await (await b.new_context(viewport={'width': width, 'height': 900})).new_page(); errs = []
        pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2000)
        await pg.evaluate("""async () => { const txt = await (await fetch('tools/_s1.json')).text();
          const db = await new Promise((res, rej) => { const r = indexedDB.open('pbl-league', 1); r.onupgradeneeded = () => r.result.createObjectStore('s'); r.onsuccess = () => res(r.result); r.onerror = () => rej(r.error); });
          const put = (k, v) => new Promise((res, rej) => { const tx = db.transaction('s', 'readwrite'); tx.objectStore('s').put(v, k); tx.oncomplete = res; tx.onerror = () => rej(tx.error); });
          await put('league:slot-t33d', txt); await put('slots', JSON.stringify([{ id: 'slot-t33d', name: 't33d', mode: 'commish', season: 11, phase: 'regular', updated: Date.now() }])); await put('current', 'slot-t33d'); db.close(); }""")
        await pg.reload(); await pg.wait_for_timeout(6000)
        W = f'{width}px: '
        await ev(pg, "document.querySelector('#modal') && (document.querySelector('#modal').innerHTML='')")
        n = await ev(pg, "Object.values(L.players).reduce((a,p)=>a+(p.hist||[]).filter(h=>h.rs&&h.rs.advEst).length,0)")
        ok(W + 'old save loaded, past sim seasons backfilled (advEst)', n > 100, str(n))
        pid = await ev(pg, "(()=>{const p=Object.values(L.players).filter(p=>!p.retired&&p.team!=null&&(p.hist||[]).filter(h=>h.rs&&h.rs.advEst&&h.rs.gp>40).length>=5).sort((a,b)=>b.ovr-a.ovr)[0];return p?p.id:null})()")
        S0 = "history.replaceState(null,'','#stats');ui.player=null;ui.game=null;ui.team=null;ui.coach=null;ui.gm=null;"
        await render(pg, f"ui.player={pid};ui.pTab={{rs:'adv',po:'adv'}}")
        t = await ev(pg, "document.querySelector('#main').innerText")
        ne = await ev(pg, "document.querySelectorAll('#main span.muted[title^=\"Estimated\"]').length")
        ok(W + 'player Advanced tab: "est." on backfilled seasons, with tooltip', 'est.' in t and ne >= 5, f'{ne} tags')
        ok(W + 'player Advanced tab: no sideways scroll', await ev(pg, "document.documentElement.scrollWidth<=window.innerWidth+2"))
        await render(pg, S0 + "history.replaceState(null,'','#history');ui.view='history';ui.histTab='records';ui.hs=null", 1500)
        t = await ev(pg, "document.querySelector('#main').innerText")
        ok(W + 'records: "(est.)" on backfilled PER / WS seasons', '(est.)' in t)
        ok(W + 'records: no sideways scroll', await ev(pg, "document.documentElement.scrollWidth<=window.innerWidth+2"))
        ok(W + 'no page errors', not errs, '; '.join(errs[:2]))
        await b.close()
async def main():
    for w in (1300, 390): await run(w)
    print('\n'.join(res)); print(f"{sum(r.startswith('PASS') for r in res)}/{len(res)} pass")
asyncio.run(main())
