# v28.4 F4 browser test: season recap on the History season page, HOF speech under inductees, no page errors. Serve the folder, then: python3 browser-f4.py PORT
import asyncio, sys
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8786'
res = []
def ok(n, c, x=''): res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else ''))
CLR = "document.querySelector('#modal').innerHTML=''"
async def main():
  async with async_playwright() as p:
    b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
    pg = await (await b.new_context(viewport={'width': 1300, 'height': 900})).new_page()
    errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
    ev = pg.evaluate
    await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
    await pg.click('[data-setmode="team"]'); await pg.click('[data-setnba="1"]'); await pg.wait_for_timeout(300)
    await pg.click('[data-setmode="team"]'); await pg.click('.teampick:has-text("Lakers")'); await pg.click('#create'); await pg.wait_for_timeout(5000); await ev(CLR)
    await ev("(()=>{const L=__pbl.L;let g=0;while(L.phase==='regular'&&g++<400)simDay(L);processPending(L);startPlayoffs(L);g=0;while((L.phase==='playin'||L.phase==='playoffs')&&g++<600)simPlayoffDay(L);startOffseason(L);__pbl.render()})()")
    s = await ev("__pbl.L.season")
    await ev(CLR); await ev(f"(()=>{{const U=__pbl.ui;U.view='history';U.histTab='seasons';U.hs={s};__pbl.render()}})()"); await pg.wait_for_timeout(200)
    r = await ev("(()=>{const h=[...document.querySelectorAll('#main h3')].find(x=>x.textContent==='Season recap');return h?h.parentElement.textContent:''})()")
    ok('season recap panel', len(r) > 80 and '{' not in r, r[:120])
    ok('recap news item', await ev("__pbl.L.log.some(x=>x.text.startsWith('Season recap: '))"))
    await ev(CLR); await ev("(()=>{const U=__pbl.ui;U.hs=null;U.histTab='awards';__pbl.render()})()"); await pg.wait_for_timeout(200)
    n = await ev("document.querySelectorAll('#main details.hofsp').length"); hof = await ev("(__pbl.L.hof||[]).length")
    ok('speech under each inductee', hof > 0 and n == hof, f'{n}/{hof}')
    if n:
      await pg.click('#main details.hofsp summary'); t = await ev("document.querySelector('#main details.hofsp p').textContent")
      ok('speech text', len(t) > 100 and '{' not in t, t[:100])
      await pg.locator('#main details.hofsp').first.screenshot(path='/tmp/f4-speech.png')
    ok('no page errors', not errs, '; '.join(errs[:3]))
    await b.close()
  for r in res: print(r)
  print(sum(r.startswith('PASS') for r in res), '/', len(res), 'pass')
asyncio.run(main())
