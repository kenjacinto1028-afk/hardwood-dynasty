import subprocess, time
from playwright.sync_api import sync_playwright
srv = subprocess.Popen(['python3','-m','http.server','8769'], cwd='/home/claude/game', stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL); time.sleep(1)
errs=[]
try:
  with sync_playwright() as pw:
    b = pw.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
    pg = b.new_page(viewport={'width':1400,'height':900}); pg.on('pageerror', lambda e: errs.append(str(e)))
    pg.goto('http://localhost:8769/index.html'); pg.wait_for_selector('#create', timeout=15000)
    pg.evaluate("(()=>{const u=__pbl.ui; u.setup=Object.assign(u.setup||{}, {mode:'team',team:0,nba:true});document.getElementById('create').click();})()"); pg.wait_for_timeout(4000)
    pg.evaluate("location.hash='#stats'"); pg.wait_for_timeout(500)
    pg.click('[data-stab="players"]'); pg.wait_for_timeout(500)
    pg.check('#pall'); pg.wait_for_timeout(500)
    rc = lambda: pg.evaluate("document.querySelector('#main table tbody').rows.length")
    print('all', rc())
    pg.select_option('#pft','13'); pg.wait_for_timeout(400); print('team13', rc())
    pg.fill('#pfa1','25'); pg.dispatch_event('#pfa1','change'); pg.wait_for_timeout(300)
    pg.fill('#pfa2','30'); pg.dispatch_event('#pfa2','change'); pg.wait_for_timeout(400)
    print('team13 age25-30', rc(), pg.evaluate("[...document.querySelector('#main table tbody').rows].map(r=>r.cells[3].textContent).join(',')"))
    pg.click('[data-padv="1"]'); pg.wait_for_timeout(400); print('adv', rc())
    pg.click('[data-pdef="1"]'); pg.wait_for_timeout(400); print('def team select', pg.evaluate("!!document.querySelector('#pft') && document.querySelector('#pft').value"), pg.evaluate("document.querySelector('#pfa1').value"))
    pg.screenshot(path='/tmp/s_stats.png')
    b.close()
finally: srv.kill()
print('errors', errs)
