# v33e browser check (1300 + 390px): retired legends' trophy case (All-NBA / All-Defense / All-Star rows from NBA_HX.aw), the single-file build,
# and the v33d floor messages ("payroll-floor deal", "Paid above his value", floor salary-dump news).
# Rename to .py, serve the game folder (python3 -m http.server PORT), then: python3 browser-v33e.py PORT [PAGE] [SAVE]
#   PAGE default index.html (use pbl-single.html after `python3 -I build.py --single .`); SAVE default tools/_s1.json (gunzip of tools/snap/nba-commish-s10.json.gz).
#   For the floor checks use a league played into free agency (SAVE=tools/_fa.json; see PROGRESS v33e).
import asyncio, sys, json
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8795'
PAGE = sys.argv[2] if len(sys.argv) > 2 else 'index.html'
SAVE = sys.argv[3] if len(sys.argv) > 3 else 'tools/_s1.json'
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui;return eval(" + json.dumps(js) + ")})()")
async def render(pg, js='0', wait=700):
    await ev(pg, "(()=>{" + js + ";window.__pbl.render()})()"); await pg.wait_for_timeout(wait)
async def run(width):
    async with async_playwright() as p:
        b = await p.chromium.launch()
        pg = await (await b.new_context(viewport={'width': width, 'height': 900})).new_page(); errs = []
        pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/{PAGE}'); await pg.wait_for_timeout(2000)
        await pg.evaluate("""async (f) => { const txt = await (await fetch(f)).text();
          const db = await new Promise((res, rej) => { const r = indexedDB.open('pbl-league', 1); r.onupgradeneeded = () => r.result.createObjectStore('s'); r.onsuccess = () => res(r.result); r.onerror = () => rej(r.error); });
          const put = (k, v) => new Promise((res, rej) => { const tx = db.transaction('s', 'readwrite'); tx.objectStore('s').put(v, k); tx.oncomplete = res; tx.onerror = () => rej(tx.error); });
          await put('league:slot-t33e', txt); await put('slots', JSON.stringify([{ id: 'slot-t33e', name: 't33e', mode: 'commish', season: 11, phase: 'regular', updated: Date.now() }])); await put('current', 'slot-t33e'); }""", SAVE)
        await pg.reload(); await pg.wait_for_timeout(6000)
        W = f'{width}px: '
        noscroll = lambda: ev(pg, "document.documentElement.scrollWidth<=window.innerWidth+2")
        await ev(pg, "document.querySelector('#modal') && (document.querySelector('#modal').innerHTML='')")
        ok(W + 'page loaded (' + PAGE + ')', await ev(pg, "typeof NBA_HX!=='undefined'&&typeof hxHonors==='function'&&typeof nbaLegends==='function'"))
        S0 = "history.replaceState(null,'','#history');ui.player=null;ui.game=null;ui.team=null;ui.coach=null;ui.gm=null;ui.view='history';ui.histTab='legends';ui.hs=null"
        await render(pg, S0, 1200)
        t = await ev(pg, "document.querySelector('#main').innerText")
        ok(W + 'Legends list shows', 'Legends' in t and 'Kareem Abdul-Jabbar' in t)
        ok(W + 'Legends list: no sideways scroll', await noscroll())
        # trophy cases: three legends with known honors (counts from NBA_HX.aw)
        for name in ['Kareem Abdul-Jabbar', 'Michael Jordan', 'Bill Russell']:
            exp = await ev(pg, f"hxHonors({json.dumps(name)})")
            await pg.click(f'tr[data-legend={json.dumps(name)}]'); await pg.wait_for_timeout(500)
            d = await ev(pg, "(()=>{const m=document.querySelector('#modal');return m?m.innerText:''})()")
            nt = await ev(pg, "document.querySelectorAll('#modal .trcase .tritem').length")
            titles = await ev(pg, "[...document.querySelectorAll('#modal .trcase .tritem')].map(e=>e.getAttribute('title'))")
            chips = await ev(pg, "document.querySelectorAll('#modal .trcase .trchip').length")
            want = sum(len(v) for v in exp.values()); kinds = len(exp)
            ok(W + name + ': trophy case rows ' + str(kinds), 'trophy case' in d.lower() and nt == kinds and kinds >= 1, f'{nt} rows, {titles}')
            ok(W + name + ': one chip per season', chips == want, f'{chips} chips vs {want}')
            ok(W + name + ': dialog fits, no sideways scroll', await noscroll() and await ev(pg, "(()=>{const t=document.querySelector('#modal .trcase');return !t||t.scrollWidth<=t.clientWidth+2})()"))
            if name == 'Michael Jordan':
                ok(W + 'Jordan: All-League 1st x10, All-Star x14 (data)', len(exp.get('m1', [])) == 10 and len(exp.get('as', [])) == 14, str({k: len(v) for k, v in exp.items()}))
                await pg.screenshot(path=f'/tmp/v33e-legend-{width}.png')
            await pg.click('#x-close'); await pg.wait_for_timeout(200)
        ok(W + 'a name with no history rows gives no honors (no empty case)', await ev(pg, "Object.keys(hxHonors('Nobody At All')).length") == 0)
        pid = await ev(pg, "(()=>{const p=Object.values(L.players).filter(p=>(p.awards||[]).length>2).sort((a,b)=>b.awards.length-a.awards.length)[0];return p?p.id:null})()")
        await render(pg, f"ui.player={pid}", 900)
        t = await ev(pg, "document.querySelector('#main').innerText")
        ok(W + 'player page still shows its trophy case', 'trophy case' in t.lower() and await ev(pg, "document.querySelectorAll('#main .trcase .tritem').length") >= 1)
        ok(W + 'player page: no sideways scroll', await noscroll())
        # v33d floor messages (only when the save has them: a league played into free agency)
        has = await ev(pg, "Object.values(L.players).some(p=>p.tx.some(x=>/payroll-floor deal/.test(x.t)))")
        if has:
            async def clipped(rx): return await pg.evaluate("(rx)=>{const re=new RegExp(rx);const bad=[];for(const e of document.querySelectorAll('#main *')){if(e.children.length===0&&re.test(e.textContent)&&(e.scrollWidth>e.clientWidth+1||e.getBoundingClientRect().right>window.innerWidth+1))bad.push(e.textContent.slice(0,60));}return bad}", rx)
            fid = await ev(pg, "Object.values(L.players).find(p=>p.tx.some(x=>/payroll-floor deal/.test(x.t))).id")
            aid = await ev(pg, "Object.values(L.players).filter(p=>p.tx.some(x=>/Paid above his value/.test(x.t))).sort((a,b)=>b.tx.length-a.tx.length)[0].id")
            for lab, pid, rx in [('payroll-floor deal', fid, 'payroll-floor deal'), ('Paid above his value', aid, 'Paid above his value')]:
                await render(pg, f"ui.view='players';ui.player={pid}", 900)
                t = await ev(pg, "document.querySelector('#main').innerText")
                ok(W + 'player page "' + lab + '" line shows', rx in t)
                ok(W + 'player page "' + lab + '": no clipped line, no sideways scroll', await noscroll() and not await clipped(rx), str(await clipped(rx)))
            await render(pg, "ui.player=null;ui.view='news';ui.ncat='trade'", 900)
            t = await ev(pg, "document.querySelector('#main').innerText")
            ok(W + 'News > Trades: floor salary-dump news shows', 'payroll floor' in t and 'Salary dump' in t)
            ok(W + 'News > Trades: no clipped line, no sideways scroll', await noscroll() and not await clipped('payroll floor'), str(await clipped('payroll floor')))
            await pg.screenshot(path=f'/tmp/v33e-news-{width}.png')
            await render(pg, "ui.view='trades';ui.trTab='done';ui.trF=''", 1000)
            t = await ev(pg, "document.querySelector('#main').innerText")
            ok(W + 'Trades history: floor salary-dump rows show', 'payroll floor' in t)
            wr = await ev(pg, "(()=>{const a=[...document.querySelectorAll('#main .whyrow')];return [a.length,a.filter(e=>e.getBoundingClientRect().right>e.closest('.tblwrap').getBoundingClientRect().right-1).length,a.filter(e=>/payroll floor/.test(e.textContent)).length]})()")
            ok(W + 'Trades history: reason / grade lines wrap inside the screen (floor dump rows included)', await noscroll() and wr[0] > 0 and wr[1] == 0 and wr[2] > 0, f'{wr[0]} rows, {wr[1]} past the table edge, {wr[2]} floor dumps')
            await pg.screenshot(path=f'/tmp/v33e-trades-{width}.png')
            await render(pg, "ui.view='home'", 900)
            ok(W + 'ticker / home: no sideways scroll with floor news in the log', await noscroll())
        else:
            print('(no floor messages in this save: floor checks skipped)', file=sys.stderr)
        ok(W + 'no page errors', not errs, '; '.join(errs[:3]))
        await b.close()
async def main():
    for w in (1300, 390): await run(w)
    print('\n'.join(res)); print(f"{sum(r.startswith('PASS') for r in res)}/{len(res)} pass")
asyncio.run(main())
