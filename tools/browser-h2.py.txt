import sys,subprocess,time
from playwright.sync_api import sync_playwright
PORT=8792
srv=subprocess.Popen(['python3','-m','http.server',str(PORT)],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL); time.sleep(1)
errs=[];res=[]
def ck(n,ok): res.append(ok); print(('PASS ' if ok else 'FAIL ')+n)
try:
  with sync_playwright() as p:
    b=p.chromium.launch(executable_path='/usr/bin/google-chrome',args=['--no-sandbox']); pg=b.new_page()
    pg.on('pageerror',lambda e:errs.append(str(e)))
    pg.goto(f'http://localhost:{PORT}/index.html'); pg.wait_for_timeout(1500)
    pg.click('[data-setmode="team"]'); pg.click('[data-setnba="1"]'); pg.click('#create'); pg.wait_for_timeout(4000)
    def show(i):
      pg.evaluate("i=>{const {ui,render}=window.__pbl;ui.coach=null;ui.gm=null;ui.view='teams';ui.team=i;ui.teamTab='overview';render()}",i)
      return pg.inner_text('.titlerib') if pg.locator('.titlerib').count() else ''
    names=pg.evaluate("()=>window.__pbl.L.teams.map(t=>t.abbr)")
    txt={n:show(i) for i,n in enumerate(names)}
    bos=txt.get('BOS',''); print(bos, [n for n,v in txt.items() if not v])
    ck('BOS 18 titles','18' in bos and 'champion' in bos.lower() and 'last 2023-24' in bos)
    ck('all 30 teams have a header line',all(txt.values()))
    ck('Finals trips shown','finals trip' in bos.lower())
    pg.evaluate("()=>{const {L}=window.__pbl;L.history.push({s:L.season,champion:5,runnerUp:6,standings:[],awards:{},leaders:{}})}")
    ck('league title adds',show(5)!=txt[names[5]] )
    ck('no page errors',not errs); print(errs[:3])
    b.close()
finally: srv.terminate()
sys.exit(0 if all(res) else 1)
