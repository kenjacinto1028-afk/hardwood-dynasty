# v30 P + G UI checks: compact philosophy panel (8 one-line rows, how-to line, tooltips, live value, AI coaches disabled, reset),
# NBA emblem logos (30 teams, small + large), PBL logos unchanged, 390px no sideways scroll, no page errors.
# Rename to .py, serve the artifact folder (python3 -m http.server PORT), then: python3 browser-v30.py PORT
import asyncio, sys
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8789'
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
CLR = "document.querySelector('#modal').innerHTML=''"
HOW = "Each slider runs 0-100 between two styles (50 = neutral). The number is his setting. The small line is his system default; far from it means he is playing against his system."
import json
# app.js is one closure: L and the logo helpers come through window.__pbl (v30 test hook); engine globals (SL_KEYS, slOf, NBA_TEXT...) are direct
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,{logo,logoBody,hexContrast,pipeLogo}=window.__pbl.v30;return eval(" + json.dumps(js) + ")})()")
async def go(pg, h):
    await ev(pg, CLR); await ev(pg, "window.__pbl.ui.player=null;window.__pbl.ui.game=null;window.__pbl.ui.coach=null"); await ev(pg, "location.hash='#standings'"); await pg.wait_for_timeout(200)
    await ev(pg, f"location.hash='#{h}'"); await pg.wait_for_timeout(700)
async def noscroll(pg): return await ev(pg, "document.documentElement.scrollWidth <= window.innerWidth + 1")
async def lineup_tab(pg):
    await ev(pg, "(()=>{const b=document.querySelector('#main [data-twin=\"rot|Lineup & rotation\"]');if(b)b.click()})()"); await pg.wait_for_timeout(600)
async def start(pg, nba):
    await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
    if nba: await pg.click('[data-setnba="1"]'); await pg.wait_for_timeout(300)
    await pg.click('[data-setmode="team"]')
    if await ev(pg, "!!document.querySelector('[data-setgames=\"58\"]')"): await pg.click('[data-setgames="58"]')
    await pg.click('#create'); await pg.wait_for_timeout(4500); await ev(pg, CLR)
async def phil_checks(pg, tag, mine):
    n = await ev(pg, "document.querySelectorAll('#phil .sl-row').length")
    ok(f'{tag}: 8 philosophy rows', n == 8, n)
    one = await ev(pg, "[...document.querySelectorAll('#phil .sl-row')].map(r=>r.offsetHeight)")
    ok(f'{tag}: every row is one line (<= 36px)', all(h <= 36 for h in one), one)
    ok(f'{tag}: each row has name, low, slider, high, value', await ev(pg, "[...document.querySelectorAll('#phil .sl-row')].every(r=>r.querySelector('.sl-nm')&&r.querySelector('.sl-lo')&&r.querySelector('input[type=range]')&&r.querySelector('.sl-hi')&&r.querySelector('.sl-val'))"))
    ok(f'{tag}: system-default tick in every slider', await ev(pg, "document.querySelectorAll('#phil .sl-rng .sl-tick').length")==8)
    ok(f'{tag}: How-to-read line', HOW in await ev(pg, "document.querySelector('#phil .sl-how').textContent"))
    ok(f'{tag}: no explanation text left in the rows', await ev(pg, "document.querySelectorAll('#phil .sl-txt').length")==0)

async def main():
  async with async_playwright() as p:
    b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
    ctx = await b.new_context(viewport={'width': 1300, 'height': 900})
    pg = await ctx.new_page(); errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
    # ================= PBL league =================
    await start(pg, False)
    await go(pg, 'myteam'); await lineup_tab(pg)
    await phil_checks(pg, 'your coach', True)
    ok('tooltip text present (NBA_TEXT.sl on each row)', await ev(pg, "SL_KEYS.every(k=>{const r=document.querySelector('#phil input[data-sl=\"'+k+'\"]').closest('.sl-row');return r.title.includes(NBA_TEXT.sl[k])&&r.title.includes('System default')})"))
    ok('all 8 rows fit without scrolling (desktop)', await ev(pg, "(()=>{const r=document.querySelector('#phil').getBoundingClientRect();return r.height<420})()"), await ev(pg, "Math.round(document.querySelector('#phil').getBoundingClientRect().height)"))
    ok('your coach: sliders enabled + reset button', await ev(pg, "[...document.querySelectorAll('#phil input[data-sl]')].every(i=>!i.disabled)") and await ev(pg, "!!document.querySelector('#sl-reset')"))
    # value updates when the slider moves
    await ev(pg, "(()=>{const i=document.querySelector('#phil input[data-sl=\"three\"]');i.value=85;i.dispatchEvent(new Event('input',{bubbles:true}))})()")
    ok('value updates when your coach slider moves', await ev(pg, "document.querySelector('#slv-three').textContent")=='85')
    await ev(pg, "document.querySelector('#phil input[data-sl=\"three\"]').dispatchEvent(new Event('change',{bubbles:true}))"); await pg.wait_for_timeout(200)
    await pg.click('#sl-reset'); await pg.wait_for_timeout(500)
    d = await ev(pg, "(()=>{const c=L.teams[L.user].coach,d=slDefault(c),s=slOf(c);return SL_KEYS.every(k=>s[k]===d[k])})()")
    ok('reset puts every slider at the system default', d)
    # AI coach: read-only
    await ev(pg, "(()=>{const t=L.teams.find(t=>t.id!==L.user);window.__pbl.ui.coach=t.coach.id;window.__pbl.render()})()"); await pg.wait_for_timeout(600)
    await phil_checks(pg, 'AI coach', False)
    ok('AI coach sliders are disabled', await ev(pg, "(()=>{const a=[...document.querySelectorAll('#phil input[data-sl]')];return a.length===8&&a.every(i=>i.disabled)})()"))
    ok('AI coach has no reset button', await ev(pg, "!document.querySelector('#sl-reset')"))
    # PBL logos unchanged (motif art, no emblem tile / wordmark band)
    pl = await ev(pg, "L.teams.map(t=>logo(t.id,64))")
    ok('PBL logos unchanged: motif badges, no emblem band', len(pl) == 30 and all('<rect x="11" y="70"' not in h and 'scale(.54)' not in h for h in pl) and await ev(pg, "L.teams.every(t=>!!t.logo)"))
    # 390px
    await pg.set_viewport_size({'width': 390, 'height': 844}); await pg.wait_for_timeout(300)
    await go(pg, 'myteam'); await lineup_tab(pg); ok('390px PBL your team (philosophy): no sideways scroll', await noscroll(pg), await ev(pg, "document.documentElement.scrollWidth"))
    ok('390px philosophy rows stay one line', await ev(pg, "[...document.querySelectorAll('#phil .sl-row')].every(r=>r.offsetHeight<=34)"))
    await ev(pg, "(()=>{const t=L.teams.find(t=>t.id!==L.user);window.__pbl.ui.coach=t.coach.id;window.__pbl.render()})()"); await pg.wait_for_timeout(500)
    ok('390px AI coach page: no sideways scroll', await noscroll(pg))
    # ================= NBA league =================
    await ctx.close(); ctx = await b.new_context(viewport={'width': 1300, 'height': 900}); pg = await ctx.new_page(); pg.on('pageerror', lambda e: errs.append(str(e)))   # fresh storage: the saved PBL league would skip the setup screen
    await start(pg, True)
    ok('NBA league has 30 teams', await ev(pg, "!!L.nba&&L.teams.length===30"))
    big = await ev(pg, "L.teams.map(t=>[t.abbr,logo(t.id,64)])"); sm = await ev(pg, "L.teams.map(t=>[t.abbr,logo(t.id,18)])")
    ok('30 large logos: emblem + wordmark band + word text', len(big) == 30 and all('scale(.54)' in h and 'textLength' in h and 'rx="5"' in h for a, h in big), [a for a, h in big if 'scale(.54)' not in h])
    ok('30 small logos: emblem only (no text)', len(sm) == 30 and all('scale(.76)' in h and '<text' not in h for a, h in sm), [a for a, h in sm if 'scale(.76)' not in h])
    ok('no initials-only fallback (abbr text never the logo body)', await ev(pg, "L.teams.every(t=>{const h=logo(t.id,64);return !h.includes('>'+t.abbr+'</text>')})"))
    ok('wordmark = NBA_TEXT.emblem word', await ev(pg, "L.teams.every(t=>logo(t.id,64).includes('>'+NBA_TEXT.emblem[t.abbr].word+'</text>'))"))
    ok('contrast kept: emblem color vs base >= 2.2 on all 30', await ev(pg, "L.teams.every(t=>{const m=logo(t.id,64).match(/<g fill=\"(#[0-9a-fA-F]{3,6})\"/);return m&&hexContrast(t.c1,m[1])>=2.2})"))
    ok('30 distinct team logos (no duplicates)', len({h for a, h in big}) == 30)
    ok('fallback: team with no emblem entry still draws initials badge', await ev(pg, "(()=>{const t=Object.assign({},L.teams[0],{abbr:'ZZZ'});const h=logoBody(t,false,64);return h.includes('>ZZZ</text>')})()"))
    await go(pg, 'standings')
    ok('standings table shows emblem logos (small)', await ev(pg, "document.querySelectorAll('#main svg.logo').length")>=30 and await ev(pg, "[...document.querySelectorAll('#main svg.logo')].every(s=>s.querySelector('g[fill]'))"))
    await ev(pg, "window.__pbl.ui.team=L.user;window.__pbl.render()"); await go(pg, 'myteam')
    ok('team page header: large emblem + word', await ev(pg, "(()=>{const s=document.querySelector('#main svg.logo[width=\"64\"]');return !!s&&!!s.querySelector('text')&&!!s.querySelector('g[fill]')})()"))
    await lineup_tab(pg); await phil_checks(pg, 'NBA your coach', True)
    # pipeline logos keep their initials badges
    ok('pipeline logos unchanged (initials badges)', await ev(pg, "(()=>{const h=pipeLogo('NCHF',0,40);return h.includes('<text')&&!h.includes('scale(.54)')})()"))
    await pg.set_viewport_size({'width': 390, 'height': 844}); await pg.wait_for_timeout(300)
    await go(pg, 'myteam'); await lineup_tab(pg); ok('390px NBA team page (large logo + philosophy): no sideways scroll', await noscroll(pg), await ev(pg, "document.documentElement.scrollWidth"))
    await go(pg, 'standings'); ok('390px NBA standings: no sideways scroll', await noscroll(pg))
    t = ''
    for v in ['myteam', 'standings', 'roster', 'news']:
        await go(pg, v); t += await ev(pg, "document.querySelector('#main').textContent")
    ok('clean text', '{' not in t and 'undefined' not in t and 'NaN' not in t)
    ok('no page errors', not errs, errs[:2])
    await b.close()
  print('\n'.join(res)); f = sum(1 for r in res if r.startswith('FAIL')); print(f'{len(res)-f} pass, {f} fail')
try:
    asyncio.run(main())
except Exception as e:
    ok('test run completed', False, str(e)[:200]); print('\n'.join(res)); f = sum(1 for r in res if r.startswith('FAIL')); print(f'{len(res)-f} pass, {f} fail')
