# v33c browser check (1300 + 390px): PER / USG% / WS / WS/48 / VORP on the player page (RS + playoffs, Advanced tab, career row),
# Stats > Players (Advanced) and Leaders, League records (season + career PER / WS / WS48), Awards tab (live ladder + last season's
# final ladder with PER / WS / BPM), History seasons list (MVP numbers). Needs a saved NBA league with 1+ sim season under v33c and
# 30 days into the next (tools: node /path/mksave.js OUT.json, see TASK_STATE v33c). Rename to .py, copy the json into the game folder
# as tools/_s1.json, serve the folder (python3 -m http.server PORT), then: python3 browser-v33c.py PORT [SHOTS_DIR]
import asyncio, sys, json
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8793'
SHOTS = sys.argv[2] if len(sys.argv) > 2 else None
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui;return eval(" + json.dumps(js) + ")})()")
async def render(pg, js='0', wait=600):
    await ev(pg, "(()=>{" + js + ";window.__pbl.render()})()"); await pg.wait_for_timeout(wait)
async def run(width):
    async with async_playwright() as p:
        try: b = await p.chromium.launch()
        except Exception: b = await p.chromium.launch(executable_path='/usr/bin/google-chrome')
        pg = await (await b.new_context(viewport={'width': width, 'height': 900})).new_page(); errs = []
        pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2000)
        await pg.evaluate("""async () => { const txt = await (await fetch('tools/_s1.json')).text();
          const db = await new Promise((res, rej) => { const r = indexedDB.open('pbl-league', 1); r.onupgradeneeded = () => r.result.createObjectStore('s'); r.onsuccess = () => res(r.result); r.onerror = () => rej(r.error); });
          const put = (k, v) => new Promise((res, rej) => { const tx = db.transaction('s', 'readwrite'); tx.objectStore('s').put(v, k); tx.oncomplete = res; tx.onerror = () => rej(tx.error); });
          await put('league:slot-t33', txt); await put('slots', JSON.stringify([{ id: 'slot-t33', name: 't33', mode: 'commish', season: 12, phase: 'regular', updated: Date.now() }])); await put('current', 'slot-t33'); db.close(); }""")
        await pg.reload(); await pg.wait_for_timeout(5000)
        W = f'{width}px: '
        noscroll = lambda: ev(pg, "document.documentElement.scrollWidth<=window.innerWidth+2")
        await ev(pg, "document.querySelector('#modal') && (document.querySelector('#modal').innerHTML='')")
        ok(W + 'saved league loaded (regular season, a finished sim season)', await ev(pg, "L.phase==='regular'&&L.day>=20&&L.history.length>=1"), str(await ev(pg, "[L.season,L.phase,L.day,L.history.length]")))
        mvp = await ev(pg, "(()=>{const s=Math.max(...Object.keys(L.awards).map(Number));return L.awards[s].mvp})()")
        main = lambda: ev(pg, "document.querySelector('#main').innerText")
        shot = lambda n: pg.screenshot(path=f'{SHOTS}/{n}-{width}.png', full_page=False) if SHOTS else asyncio.sleep(0)
        S0 = "history.replaceState(null,'','#stats');ui.player=null;ui.game=null;ui.team=null;ui.coach=null;ui.gm=null;"
        for view, js, need in [
          ('player page advanced (RS + PO)', f"ui.player={mvp};ui.pTab={{rs:'adv',po:'adv'}}", ['PER', 'OWS', 'DWS', 'WS/48', 'VORP', 'Career', 'Playoffs']),
          ('player page per game', f"ui.player={mvp};ui.pTab={{rs:'pg',po:'pg'}}", ['USG%', 'BPM']),
          ('stats players advanced', S0 + "ui.view='stats';ui.statTab='players';ui.pAdv=true;ui.pDef=false", ['PER', 'OWS', 'DWS', 'WS/48', 'VORP']),
          ('stats players basic', S0 + "ui.view='stats';ui.statTab='players';ui.pAdv=false;ui.pDef=false", ['PER', 'WS']),
          ('leaders', S0 + "ui.view='stats';ui.statTab='leaders'", ['PER', 'Win shares', 'Win shares per 48', 'VORP', 'Defensive win shares']),
          ('league records', S0 + "history.replaceState(null,'','#history');ui.view='history';ui.histTab='records';ui.hs=null", ['PER · season', 'Win shares · season', 'Win shares per 48 · season', 'PER · career', 'Win shares · career', 'Wilt Chamberlain']),
          ('awards live ladder', S0 + "history.replaceState(null,'','#awards');ui.view='awards';ui.awKey='mvp';ui.awSeason=null", ['PER', 'WS', 'BPM', 'WS/48', 'Live race']),
          ('awards DPOY ladder', S0 + "ui.view='awards';ui.awKey='dpoy';ui.awSeason=null", ['DWS', 'DBPM']),
          ('awards last season', S0 + "ui.view='awards';ui.awKey='mvp';ui.awSeason=L.season-1", ['Final results', 'PER', 'WS', 'Winner']),
          ('history seasons', S0 + "history.replaceState(null,'','#history');ui.view='history';ui.histTab='seasons';ui.hs=null", ['PER', 'WS', 'BPM', '2015-16'])]:
            await render(pg, js, 1000); t = await main()
            has = lambda x: x in t or (len(x) > 4 and x.lower() in t.lower())   # headings are upper-cased by CSS (innerText); short codes must match exactly
            ok(W + view + ': shows ' + ', '.join(need), all(has(x) for x in need), ','.join(x for x in need if not has(x)))
            ok(W + view + ': no horizontal page scroll', await noscroll())
            await shot(view.replace(' ', '-').replace('(', '').replace(')', '').replace('+', ''))
        # numbers, not dashes: the MVP's PER / WS on his page and in the ladder
        await render(pg, f"ui.player={mvp};ui.pTab={{rs:'adv',po:'adv'}}", 800)
        nums = await ev(pg, f"(()=>{{const p=L.players[{mvp}],h=p.hist[p.hist.length-1];return [h.rs.per,h.rs.ws,h.rs.ws48,h.rs.vorp,p.stats.rs?p.stats.rs.per:null]}})()")
        ok(W + 'MVP has PER / WS / WS48 / VORP last season and PER this season', all(isinstance(x, (int, float)) for x in nums), str(nums))
        ok(W + 'no page errors', not errs, '; '.join(errs[:3]))
        await b.close()
async def main():
    for w in (1300, 390): await run(w)
    print('\n'.join(res)); print(f"{sum(r.startswith('PASS') for r in res)}/{len(res)} pass")
asyncio.run(main())
