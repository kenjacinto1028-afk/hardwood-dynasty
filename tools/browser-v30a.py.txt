res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
CLR = "document.querySelector('#modal').innerHTML=''"
def clean(t): return '{' not in t and 'undefined' not in t and 'NaN' not in t
# app.js is one closure: L / ui / render come through window.__pbl; engine globals (simDay, twWillSign, faRoom...) are direct
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui;return eval(" + json.dumps(js) + ")})()")
async def tc(pg, sel='#main'): return await ev(pg, f"(document.querySelector('{sel}')||{{textContent:''}}).textContent")
async def go(pg, h):
    await ev(pg, CLR); await ev(pg, "ui.player=null;ui.game=null;ui.coach=null"); await ev(pg, "location.hash='#standings'"); await pg.wait_for_timeout(200)
    await ev(pg, f"location.hash='#{h}'"); await pg.wait_for_timeout(700)
async def player(pg, pid):
    await ev(pg, CLR); await ev(pg, f"ui.player={pid};window.__pbl.render()"); await pg.wait_for_timeout(500)
async def noscroll(pg): return await ev(pg, "document.documentElement.scrollWidth <= window.innerWidth + 1")
async def narrow(pg, label, js_before=None):
    """same view at 390px: no sideways scroll, no stray template text"""
    await pg.set_viewport_size({'width': 390, 'height': 844}); await pg.wait_for_timeout(400)
    await ev(pg, "window.__pbl.render()"); await pg.wait_for_timeout(300)
    t = await tc(pg); ok(f'390px: {label}: no sideways scroll', await noscroll(pg), await ev(pg, "document.documentElement.scrollWidth"))
    ok(f'390px: {label}: clean text', clean(t))
    await pg.set_viewport_size({'width': 1300, 'height': 900}); await pg.wait_for_timeout(300)
async def close_dialogs(pg, seen):
    for _ in range(8):
        h = await ev(pg, "(()=>{const h=document.querySelector('#modal h2');return h?h.textContent.trim():null})()")
        if h is None: break
        seen.append(h)
        await ev(pg, "(()=>{const b=document.querySelector('#modal #x-close')||document.querySelector('#modal .dx');if(b)b.click()})()"); await pg.wait_for_timeout(250)
async def click_sim(pg, k, seen):
    ok(f'sim button "{k}" is available', await ev(pg, f"!!document.querySelector('#simctl [data-sim=\"{k}\"]')"))
    await ev(pg, f"document.querySelector('#simctl [data-sim=\"{k}\"]').click()"); await pg.wait_for_timeout(2500)
    await close_dialogs(pg, seen)
