# v28.7 checks: autopilot (Sim seasons), locker room drama UI, Dynasties & eras tab, mobile layout. Serve the artifact folder, then: python3 browser-v287.py PORT
import asyncio, sys, re
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8787'
res = []
def ok(n, c, x=''): res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else ''))
CLR = "document.querySelector('#modal').innerHTML=''"
async def ev(pg, js): return await pg.evaluate(js)
async def go(pg, h):
    await ev(pg, CLR); await ev(pg, "location.hash='#standings'"); await pg.wait_for_timeout(200)
    await ev(pg, f"location.hash='#{h}'"); await pg.wait_for_timeout(700)
async def tc(pg, sel='#main'): return await ev(pg, f"(document.querySelector('{sel}')||{{textContent:''}}).textContent")
def clean(t): return '{' not in t and 'undefined' not in t and 'NaN' not in t
async def main():
  async with async_playwright() as p:
    b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
    pg = await (await b.new_context(viewport={'width': 1300, 'height': 900})).new_page()
    errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
    await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
    await pg.click('[data-setmode="team"]'); await pg.click('[data-setgames="58"]'); await pg.click('#create'); await pg.wait_for_timeout(4500); await ev(pg, CLR)
    u = await ev(pg, "window.__pbl.L.user")
    ok('Sim seasons button in header', await ev(pg, "!!document.querySelector('#simctl [data-sim=\"multi\"]')"))
    await pg.click('#simctl [data-sim="multi"]'); await pg.wait_for_timeout(300)
    t = await tc(pg, '#modal'); ok('dialog explains the AI runs your team', 'Sim seasons' in t and 'does not stop' in t, t[:80])
    await pg.click('[data-autn="1"]'); await pg.click('#aut-go'); await pg.wait_for_timeout(1500)
    ok('Stop button while running', await ev(pg, "!!document.querySelector('#simctl [data-sim=\"autostop\"]')"))
    ok('header shows autopilot', 'Autopilot' in await tc(pg, '#when'), await tc(pg, '#when'))
    ok('your team handed to AI while running', await ev(pg, "window.__pbl.L.user===null && !!window.__pbl.L.auto"))
    for i in range(300):
        if not await ev(pg, "window.__pbl.ui.auto"): break
        await pg.wait_for_timeout(1000)
    t = await tc(pg, '#modal')
    ok('summary dialog', 'Autopilot finished' in t and 'What the AI did' in t and clean(t), t[:120])
    ok('team given back', await ev(pg, f"window.__pbl.L.user==={u} && !window.__pbl.L.auto && window.__pbl.L.phase==='off-ready'"), await ev(pg, "window.__pbl.L.phase"))
    ok('summary has a season row with W-L', re.search(r'\d+-\d+', t) is not None)
    await ev(pg, CLR)
    # dynasties tab
    await go(pg, 'history'); await ev(pg, "window.__pbl.ui.histTab='dyn';window.__pbl.render()"); await pg.wait_for_timeout(300)
    t = await tc(pg); ok('Dynasties & eras tab', 'Dynasty watch' in t and 'Eras' in t and clean(t), t[:100])
    # start next season, sim a few days, force a locker room issue on your team
    await ev(pg, "document.querySelector('#simctl [data-sim=\"newseason\"]').click()"); await pg.wait_for_timeout(4000); await ev(pg, CLR)
    await ev(pg, "(()=>{const L=window.__pbl.L;for(let i=0;i<10;i++)simDay(L);const t=L.teams[L.user];const r=rotationOf(L,t);const e={id:'drT',s:L.season,day:L.day,tid:t.id,k:'feud',pids:[r[0].id,r[1].id],st:'open',text:r[0].name+' and '+r[1].name+' had to be separated at practice.'};L.drama=(L.drama||[]);L.drama.unshift(e);L.dramaOpen='drT';window.__pbl.render()})()")
    await pg.wait_for_timeout(300)
    ok('header Locker room issue button', await ev(pg, "!!document.querySelector('#simctl [data-sim=\"drama\"]')"))
    await go(pg, 'myteam'); t = await tc(pg); ok('locker room panel on My team', 'Locker room' in t and 'Decide' in t and clean(t))
    await pg.click('#simctl [data-sim="drama"]'); await pg.wait_for_timeout(300)
    t = await tc(pg, '#modal'); ok('drama dialog with 5 choices', await ev(pg, "document.querySelectorAll('[data-drc]').length")==5 and 'separated' in t)
    await pg.click('[data-drc="talk"]'); await pg.wait_for_timeout(300)
    t = await tc(pg, '#modal'); ok('result shows effects', 'Locker room' in t and 'morale' in t and clean(t), t[:120])
    ok('issue closed', await ev(pg, "!window.__pbl.L.dramaOpen && window.__pbl.L.drama[0].choice==='talk'"))
    await ev(pg, CLR)
    await ev(pg, "(()=>{const L=window.__pbl.L;updateMorale(L);const p=rosterOf(L,L.teams[L.user])[0];window.__pbl.ui.player=p.id;window.__pbl.render()})()"); await pg.wait_for_timeout(400)
    t = await tc(pg); ok('morale panel shows Locker room row', 'Locker room' in t or 'Feud with a teammate' in t or 'Morale' in t)
    await ev(pg, "window.__pbl.ui.player=null;window.__pbl.render()")
    # mobile
    m = await (await b.new_context(viewport={'width': 390, 'height': 844}, is_mobile=True, has_touch=True)).new_page()
    m.on('pageerror', lambda e: errs.append('mobile: ' + str(e)))
    await m.goto(f'http://localhost:{PORT}/index.html'); await m.wait_for_timeout(2500)
    await m.click('[data-setmode="team"]'); await m.click('[data-setgames="58"]'); await m.click('#create'); await m.wait_for_timeout(4500); await m.evaluate(CLR)
    for h in ['myteam', 'standings', 'stats', 'trades', 'history', 'feed']:
        await m.evaluate(CLR); await m.evaluate(f"location.hash='#{h}'"); await m.wait_for_timeout(700)
        w = await m.evaluate("document.documentElement.scrollWidth")
        ok(f'mobile {h}: no sideways page scroll', w <= 392, w)
        await m.screenshot(path=f'/tmp/m-{h}.png')
    await m.evaluate(CLR); await m.click('#simctl [data-sim="multi"]'); await m.wait_for_timeout(300)
    box = await m.evaluate("(()=>{const d=document.querySelector('.dialog').getBoundingClientRect();return [d.left,d.right,d.bottom]})()")
    ok('mobile dialog fits the screen', box[0] >= 0 and box[1] <= 391 and box[2] <= 845, box)
    await m.screenshot(path='/tmp/m-dialog.png')
    ok('no page errors', not errs, errs[:3])
    await b.close()
  print('\n'.join(res)); print(sum(r.startswith('PASS') for r in res), 'pass,', sum(r.startswith('FAIL') for r in res), 'fail')
asyncio.run(main())
