# v28.5 F5 browser test: Regular season / Playoffs toggle in "Real NBA seasons", ~ marks, playoff records, no page errors. Serve the folder, then: python3 browser-f5.py PORT
import asyncio, sys
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8786'
res = []
def ok(n, c, x=''): res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else ''))
CLR = "document.querySelector('#modal').innerHTML=''"
async def main():
  async with async_playwright() as p:
    b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
    pg = await (await b.new_context(viewport={'width': 1300, 'height': 900})).new_page()
    errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
    ev = pg.evaluate
    await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
    await pg.click('[data-setmode="team"]'); await pg.click('[data-setnba="1"]'); await pg.wait_for_timeout(300)
    await pg.click('[data-setmode="team"]'); await pg.click('.teampick:has-text("Lakers")'); await pg.click('#create'); await pg.wait_for_timeout(5000); await ev(CLR)
    async def show(name, tab):
      await ev(f"(()=>{{const U=__pbl.ui,L=__pbl.L;const p=Object.values(L.players).find(x=>x.name==='{name}');U.view='player';U.player=p.id;U.pTab={{rs:'{tab}'}};U.pSet='rs';__pbl.render()}})()"); await pg.wait_for_timeout(250)
    await show('LeBron James', 'tot')
    h = await ev("(()=>{const x=[...document.querySelectorAll('#main h4')].find(h=>h.textContent.startsWith('Real NBA'));return x?x.textContent:''})()")
    ok('toggle shown on Real NBA seasons', 'Playoffs' in h and 'Regular season' in h, h)
    rs = await ev("(()=>{const x=[...document.querySelectorAll('#main h4')].find(h=>h.textContent.startsWith('Real NBA'));return x.parentElement.querySelectorAll('tbody tr').length})()")
    await pg.click('[data-pset="po"]'); await pg.wait_for_timeout(250)
    h2 = await ev("(()=>{const x=[...document.querySelectorAll('#main h4')].find(h=>h.textContent.startsWith('Real NBA'));return x.textContent})()")
    t = await ev("(()=>{const x=[...document.querySelectorAll('#main h4')].find(h=>h.textContent.startsWith('Real NBA'));const tb=x.nextElementSibling;return tb.textContent})()")
    ok('playoffs table', 'playoffs' in h2 and '2011-12' in t and '2025-26' in t, h2)
    # career row pts
    pts = await ev("(()=>{const x=[...document.querySelectorAll('#main h4')].find(h=>h.textContent.startsWith('Real NBA'));const r=[...x.nextElementSibling.querySelectorAll('tbody tr')].pop();return r.textContent})()")
    ok('career playoff row has 8,521 pts total', '8521' in pts.replace(',', '') or '8,521' in pts, pts[:80])
    await pg.click('[data-ptab="rs:adv"]'); await pg.wait_for_timeout(250)
    ap = await ev("document.querySelectorAll('#main span[title=\"estimated from box score\"]').length")
    ok('~ marks with tooltip on advanced playoff values', ap > 20, str(ap))
    await pg.click('[data-pset="rs"]'); await pg.wait_for_timeout(250)
    ap2 = await ev("document.querySelectorAll('#main span[title=\"estimated from box score\"]').length")
    ok('LeBron real RS advanced has no ~', ap2 == 0, str(ap2))
    await show('Marcus Smart', 'adv')
    ap3 = await ev("document.querySelectorAll('#main span[title=\"estimated from box score\"]').length")
    est = await ev("document.querySelectorAll('#main .est-tag').length")
    ok('role player: real rows, ~ on advanced, no est tags', ap3 > 20 and est == 0, f'{ap3} ~, {est} est')
    await ev("(()=>{const U=__pbl.ui;U.player=null;U.view='history';U.histTab='records';__pbl.render()})()"); await pg.wait_for_timeout(400)
    txt = await ev("document.querySelector('#main').textContent")
    ok('career playoff records on League records', 'Points · career playoffs' in txt and 'LeBron James' in txt)
    ok('no page errors', not errs, '; '.join(errs[:3]))
    await pg.screenshot(path='/tmp/f5-records.png')
    await b.close()
  for r in res: print(r)
  print(sum(r.startswith('PASS') for r in res), '/', len(res), 'pass')
asyncio.run(main())
