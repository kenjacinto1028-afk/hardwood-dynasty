# v26.6 Sonnet-part browser checks (S1-S5). Serve the artifact folder, then: python3 browser-v266.py PORT
import asyncio, sys
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8778'
res = []
def ok(n, c, x=''): res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else ''))
CLR = "document.querySelector('#modal').innerHTML=''"
async def go(pg, h):
    await pg.evaluate(CLR); await pg.evaluate("location.hash='#standings'"); await pg.wait_for_timeout(200)
    await pg.evaluate(f"location.hash='#{h}'"); await pg.wait_for_timeout(700)
async def txt(pg): return (await pg.evaluate("document.querySelector('#main').innerText"))
async def tc(pg): return await pg.evaluate("document.querySelector('#main').textContent")
def clean(t): return '{' not in t and 'undefined' not in t and 'NaN' not in t
async def ev(pg, js): return await pg.evaluate(js)
async def main():
  try:
    async with async_playwright() as p:
        b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
        pg = await (await b.new_context(viewport={'width': 1300, 'height': 900})).new_page()
        errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
        # S1 setup screen
        ok('S1 three season lengths on the setup screen', await ev(pg, "[...document.querySelectorAll('[data-setgames]')].map(b=>b.dataset.setgames).join()==='82,66,58'"))
        ok('S1 82 is the default', await ev(pg, "document.querySelector('[data-setgames=\"82\"]').classList.contains('on')"))
        await pg.click('[data-setgames="58"]'); ok('S1 pick 58', await ev(pg, "document.querySelector('[data-setgames=\"58\"]').classList.contains('on')") and '58-game season' in (await txt(pg)).lower())
        await pg.click('[data-setgames="82"]'); await pg.click('[data-setmode="team"]'); await pg.click('#create'); await pg.wait_for_timeout(4500); await ev(pg, CLR)
        ok('S1 league made at 82 games', await ev(pg, "window.__pbl.L.sg===82&&SEASON_GAMES===82&&TW_GAMES===28"))
        await go(pg, 'schedule'); ok('S1 schedule says 82-game season', '82-game season' in (await txt(pg)).lower() and clean(await tc(pg)))
        await go(pg, 'preview'); ok('S1 preview says 82-game season', '82-game season' in (await txt(pg)).lower())
        await go(pg, 'rules'); t = await tc(pg); ok('S1 rules show 82 games', 'plays 82 games' in t)
        await ev(pg, "(()=>{const L=window.__pbl.L;for(let i=0;i<45;i++)simDay(L);window.__pbl.render()})()")
        for v in ['standings', 'stats', 'teams', 'injuries']:
            await go(pg, v); ok(f'S1 {v} page clean at 82', clean(await tc(pg)))
        ok('S1 two-way view', await ev(pg, "(()=>{const L=window.__pbl.L;const p=Object.values(L.players).find(x=>x.tw&&!x.retired);if(!p)return true;window.__pbl.ui.player=p.id;window.__pbl.render();const t=document.querySelector('#main').textContent;window.__pbl.ui.player=null;return t.includes('of 28 games')||t.includes('/28 games')||t.includes('/28')})()"))
        await ev(pg, "window.__pbl.render()")
        # season leaders thresholds scale
        ok('S1 gpq thresholds', await ev(pg, "gpq(29)===41&&gpq(20)===28&&gpq(15)===21"))
        # S2 development role
        await go(pg, 'myteam')
        ok('S2 force a development role', await ev(pg, "(()=>{const L=window.__pbl.L,t=L.teams[L.user];t.status='rebuild';t.mode='rebuild';const p=rosterOf(L,t).slice().sort((a,b)=>a.age-b.age)[0];p.age=21;p.scout=84;p.ovr=64;window.__pid=p.id;const d=devRole(L,t,p);window.__pbl.render();return !!d&&d.pot>0})()"))
        ok('S2 pill on the roster with tooltip', await ev(pg, "(()=>{const e=document.querySelector('#main .pill.dev-role');return !!e&&e.title.includes('potential +')&&e.title.includes('role adapted +')})()"))
        await ev(pg, "(()=>{window.__pbl.ui.player=window.__pid;window.__pbl.render()})()"); await pg.wait_for_timeout(400); t = await txt(pg)
        ok('S2 pill on the player page', await ev(pg, "!!document.querySelector('#main .pill.dev-role')"))
        ok('S2 coach fit line under Traits', 'coach fit' in t.lower() and '%' in t and clean(await tc(pg)))
        await ev(pg, "(()=>{window.__pbl.ui.player=null;const L=window.__pbl.L;L.teams[L.user].status=null;window.__pbl.render()})()")
        ok("S2 Doesn't fit system pill", await ev(pg, "(()=>{const L=window.__pbl,l=L.L;const t=l.teams.find(t=>t.id!==l.user);const p=rosterOf(l,t)[5];t.tstat=t.tstat||{};t.tstat[p.id]={k:'block',why:\"Doesn't fit the coach's system\"};L.ui.view='teams';L.ui.team=t.id;L.ui.mtab='roster';L.render();return !!document.querySelector('#main .pill.dev-nofit')})()"))
        # S3 FA media
        await ev(pg, "(()=>{const l=window.__pbl.L;const ids=Object.keys(l.players).slice(0,3);l.rumors=l.rumors||[];l.rumors.unshift({n:99001,s:l.season,day:null,pid:+ids[0],from:null,to:null,k:'fa',real:1,fa:[1,2],hit:1},{n:99002,s:l.season,day:null,pid:+ids[1],from:null,to:null,k:'fa',real:1,fa:[3],hit:-1},{n:99003,s:l.season,day:null,pid:+ids[2],from:null,to:null,k:'fa',real:1,fa:[4]});l.faWhy=l.faWhy||{};l.faWhy[ids[0]]={s:l.season,tid:1,why:'money'};l.faWhy[ids[1]]={s:l.season,tid:2,why:'loyal'};})()")
        await go(pg, 'fa'); t = await txt(pg)
        ok('S3 Around the league box', 'around the league' in t.lower())
        ok('S3 rumors with Right / Wrong / Open', all(x in t for x in ['Right', 'Wrong', 'Open']) and 'meeting with' in t)
        ok('S3 decisions with reason', 'Paid the most' in t and 'Loyalty' in t)
        ok('S3 FA page clean', clean(await tc(pg)))
        # S4 hot seat, fan mood, new GM
        await ev(pg, "(()=>{const l=window.__pbl.L;l.teams[l.user].hot=55})()"); await go(pg, 'myteam')
        ok('S4 hot seat badge at 55 (not red)', await ev(pg, "(()=>{const e=document.querySelector('#main .pill.hot-seat');return !!e&&!e.classList.contains('hot-hi')&&e.title.includes('55')})()"))
        await ev(pg, "(()=>{const l=window.__pbl.L;l.teams[l.user].hot=75;l.teams[l.user].coach.stints&&0;window.__pbl.render()})()")
        ok('S4 red at 75 with reasons', await ev(pg, "!!document.querySelector('#main .pill.hot-seat.hot-hi')"))
        await ev(pg, "(()=>{const l=window.__pbl.L;window.__pbl.ui.coach=l.teams[l.user].coach.id;window.__pbl.render()})()"); await pg.wait_for_timeout(400)
        ok('S4 hot seat badge on the coach page', await ev(pg, "!!document.querySelector('#main .pill.hot-seat')"))
        await ev(pg, "(()=>{const l=window.__pbl.L;window.__pbl.ui.coach=null;l.teams[l.user].hot=10;window.__pbl.render()})()")
        ok('S4 no badge under 50', await ev(pg, "!document.querySelector('#main .pill.hot-seat')"))
        await go(pg, 'myteam'); await pg.click('[data-mtab="office"]'); await pg.wait_for_timeout(400); t = await txt(pg)
        ok('S4 fan mood meter', 'fan mood' in t.lower() and await ev(pg, "!!document.querySelector('#main .fan-bar')"))
        await ev(pg, "(()=>{const l=window.__pbl.L;const t=l.teams.find(t=>t.id!==l.user);t.newGM=l.season;t.lowRun=2;window.__t=t.id;window.__pbl.ui.view='teams';window.__pbl.ui.team=t.id;window.__pbl.ui.mtab='office';window.__pbl.render()})()"); await pg.wait_for_timeout(300)
        t = await txt(pg)
        ok('S4 New GM chip and low-seasons line on an AI team', await ev(pg, "!!document.querySelector('#main .pill.new-gm')") and 'low seasons in a row' in t.lower())
        ok('S4 office page clean', clean(await tc(pg)))
        # S5 storylines
        await ev(pg, "(()=>{const l=window.__pbl.L;window.__pbl.ui.team=null;window.__pbl.ui.view='myteam';window.__pbl.ui.mtab='roster';logNews(l,'Test Player returns to face his old team','return'===0?0:1,'feature','return');logNews(l,'Test Player is the new MVP favorite',1,'feature','breakout');addPost(l,'race','nat','calm',{name:'Test Player',team:'Test',prev:'Someone',pts:'30'},{tid:1,size:1.5});window.__pbl.render()})()")
        await go(pg, 'myteam'); t = await txt(pg)
        ok('S5 Storylines box on My team', 'storylines' in t.lower() and 'returns to face his old team' in t, ('STORYLINES' in t, 'old team' in t))
        ok('S5 shows at most 5 news + 3 posts', await ev(pg, "[...document.querySelectorAll('#main .panel')].find(p=>/storylines/i.test(p.querySelector('h3')?.textContent||'')).querySelectorAll('.leader').length<=5"))
        ok('S5 My team clean', clean(await tc(pg)), __import__('re').findall(r'.{40}(?:\{|undefined|NaN).{20}', await tc(pg))[:2])
        # commissioner league at 58 games
        await ev(pg, "(()=>{window.__pbl.ui.setupOpen=true;window.__pbl.render()})()"); await pg.wait_for_timeout(400)
        await pg.click('[data-setgames="58"]'); await pg.click('[data-setmode="commish"]'); await pg.click('#create'); await pg.wait_for_timeout(4500); await ev(pg, CLR)
        ok('S1 58-game league', await ev(pg, "window.__pbl.L.sg===58&&SEASON_GAMES===58&&TW_GAMES===20"))
        await go(pg, 'schedule'); ok('S1 schedule says 58-game season', '58-game season' in (await txt(pg)).lower())
        await ev(pg, "(()=>{const L=window.__pbl.L;for(let i=0;i<20;i++)simDay(L);window.__pbl.render()})()")
        await go(pg, 'standings'); ok('S5 commissioner standings clean', clean(await tc(pg)))
        ok('no page errors', not errs, '; '.join(errs)[:300])
        await b.close()
  finally:
    print('\n'.join(res)); print(sum(r.startswith('PASS') for r in res), '/', len(res))
asyncio.run(main())
