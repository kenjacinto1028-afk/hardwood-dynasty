# v32b/v32c browser check: roster Role column, player Role box, offer Role picker (promise changes the ask), Rotation & roles panel,
# live game sub lines + starting lineup window from the timeline. Desktop 1300 and 390px.
# Rename to .py, serve the game folder (python3 -m http.server PORT), then: python3 browser-v32bc.py PORT
import asyncio, sys, json
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8792'
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui;return eval(" + json.dumps(js) + ")})()")
async def run(width):
    async with async_playwright() as p:
        b = await p.chromium.launch(); pg = await (await b.new_context(viewport={'width': width, 'height': 900})).new_page(); errs = []
        pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
        await pg.click('[data-setmode="team"]'); await pg.click('[data-setnba="1"]'); await pg.click('[data-setgames="58"]'); await pg.click('#create'); await pg.wait_for_timeout(4500)
        await ev(pg, "document.querySelector('#modal').innerHTML=''")
        W = f'{width}px: '
        # roster role column
        await ev(pg, "(()=>{ui.player=null;ui.game=null;ui.view='myteam';ui.mtab='roster';window.__pbl.render()})()"); await pg.wait_for_timeout(500)
        cells = await ev(pg, "(()=>{const t=[...document.querySelectorAll('table[data-sid=roster] thead th')].map(x=>x.textContent);const i=t.indexOf('Role');if(i<0)return null;return [...document.querySelectorAll('table[data-sid=roster] tbody tr')].map(r=>r.children[i]&&r.children[i].textContent)})()")
        ok(W + 'roster has a Role column with a role for every player', bool(cells) and all(any(k in c for k in ['Franchise star', 'Star', 'Starter', 'Sixth man', 'Rotation', 'Bench', 'Development']) for c in cells), json.dumps((cells or [])[:4]))
        # rotation panel role names
        rp = await ev(pg, "[...document.querySelectorAll('table[data-sid=rotation] tbody tr td:first-child')].map(x=>x.textContent).slice(0,7)")
        ok(W + 'Rotation & roles panel uses role names', any('Sixth man' in x for x in rp) and any('(' in x for x in rp), json.dumps(rp))
        # player page role box
        pid = await ev(pg, "(()=>{const t=L.teams[L.user];return t.roster.map(id=>L.players[id]).sort((a,b)=>b.ovr-a.ovr)[2].id})()")
        await ev(pg, f"(()=>{{ui.view=null;ui.player={pid};window.__pbl.render()}})()"); await pg.wait_for_timeout(500)
        box = await ev(pg, "(()=>{const k=[...document.querySelectorAll('.sumcard .k')].find(x=>x.textContent==='Role');return k?k.nextElementSibling.textContent:null})()")
        ok(W + 'player summary shows his role and minutes range', bool(box) and 'min' in box, box or '')
        # offer form: free agent, role picker
        fa = await ev(pg, "(()=>{L.phase='regular';const f=L.fa.map(id=>L.players[id]).filter(p=>p&&p.ovr>=55).sort((a,b)=>b.ovr-a.ovr)[0];return f?f.id:null})()")
        if fa is not None:
            await ev(pg, f"(()=>{{ui.player={fa};window.__pbl.render()}})()"); await pg.wait_for_timeout(500)
            has = await ev(pg, "!!document.querySelector('#negbar [data-nego]')")
            if has:
                await pg.click('#negbar [data-nego]'); await pg.wait_for_timeout(600)
                opts = await ev(pg, "(()=>{const s=document.querySelector('#neg-role');return s?[...s.options].map(o=>o.textContent):null})()")
                ok(W + 'offer form has a Role picker (coach decides + 7 promises)', bool(opts) and len(opts) == 8 and opts[0].startswith('Coach decides'), json.dumps((opts or [])[:3]))
                q = await ev(pg, f"(()=>{{const t=L.teams[L.user],p=L.players[{fa}],er=expRole(L,t,p),up=ROLES[Math.max(0,roleRank(er)-1)].k;const a=negotiate(L,p.id,MINSAL,2,L.user,'signing',null,{{quote:true}}),b=negotiate(L,p.id,MINSAL,2,L.user,'signing',null,{{quote:true,role:{{k:up}}}});return [er,up,a.quote,b.quote]}})()")
                ok(W + 'promising a bigger role lowers his quote', q[3] is not None and q[2] is not None and q[3] <= q[2], json.dumps(q))
                await ev(pg, "(()=>{const s=document.querySelector('#neg-role');s.value='starter';s.dispatchEvent(new Event('change',{bubbles:true}))})()"); await pg.wait_for_timeout(400)
                kept = await ev(pg, f"(()=>{{const n=ui.neg[{fa}]||{{}};return n.role}})()")
                ok(W + 'picked role is kept in the offer state', kept == 'starter', str(kept))
            else: ok(W + 'free agent has a Negotiate button', False)
        await ev(pg, "document.querySelector('#modal').innerHTML=''")
        # play games, open the user's last game live
        await ev(pg, "(()=>{L.phase='regular';for(let d=0;d<4;d++)simDay(L);ui.player=null;window.__pbl.render()})()"); await pg.wait_for_timeout(400)
        gid = await ev(pg, "(()=>{const g=L.games.filter(x=>x.pbp&&(x.home===L.user||x.away===L.user)).slice(-1)[0];return g?g.id:null})()")
        ok(W + 'your game has play-by-play and a timeline', gid is not None and await ev(pg, f"!!L.games.find(g=>g.id==={gid}).tl[0]"))
        if gid is not None:
            await ev(pg, f"(()=>{{ui.game={gid};ui.player=null;ui.liveMode={gid};ui.lv=null;window.__pbl.render()}})()"); await pg.wait_for_timeout(600)
            st = await ev(pg, f"(()=>{{const g=L.games.find(x=>x.id==={gid});const on=[...document.querySelectorAll('.lv-court-tm .lv-cpn a')].map(a=>+a.dataset.player).sort().join();return [on,[...g.tl[0].st,...g.tl[1].st].sort().join()]}})()")
            ok(W + 'starting lineup window = timeline starters (before tip-off)', st[0] == st[1] and len(st[0]) > 0, st[0][:40])
            await ev(pg, f"(()=>{{const g=L.games.find(x=>x.id==={gid});ui.lv.i=g.pbp.length;ui.lv.paused=true;window.__pbl.render()}})()"); await pg.wait_for_timeout(600)
            subs = await ev(pg, "[...document.querySelectorAll('#lv-feed .lv-tx')].map(x=>x.textContent).filter(t=>/ sub: /.test(t)).length")
            ok(W + 'live feed shows sub lines', subs > 5, str(subs))
            sw = await ev(pg, "document.documentElement.scrollWidth<=window.innerWidth+2")
            ok(W + 'live page has no horizontal page scroll', sw)
        ok(W + 'no page errors', not errs, '; '.join(errs[:2]))
        await b.close()
async def main():
    await run(1300); await run(390)
    print('\n'.join(res)); print(f"\nbrowser-v32bc: {sum(r.startswith('PASS') for r in res)}/{len(res)}")
asyncio.run(main())
