# v32d browser check (desktop 1300 and 390px): rotation chart on the game page, planned subs in the live window, rotation chart editor,
# roles + expected minutes (roster, player page, offer form), mid-level box in the offer popup and the cap box, coaching targets cap-fit column,
# Draft tab > Team picks. Rename to .py, serve the game folder (python3 -m http.server PORT), then: python3 browser-v32d.py PORT
import asyncio, sys, json
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8793'
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui;return eval(" + json.dumps(js) + ")})()")
async def render(pg, js='0', wait=500):
    await ev(pg, "(()=>{" + js + ";window.__pbl.render()})()"); await pg.wait_for_timeout(wait)
async def run(width):
    async with async_playwright() as p:
        b = await p.chromium.launch(); pg = await (await b.new_context(viewport={'width': width, 'height': 900})).new_page(); errs = []
        pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
        await pg.click('[data-setmode="team"]'); await pg.click('[data-setnba="1"]'); await pg.click('[data-setgames="58"]'); await pg.click('#create'); await pg.wait_for_timeout(4500)
        await ev(pg, "document.querySelector('#modal').innerHTML=''")
        W = f'{width}px: '
        noscroll = lambda: ev(pg, "document.documentElement.scrollWidth<=window.innerWidth+2")
        await ev(pg, "(()=>{L.phase='regular';for(let d=0;d<6;d++)simDay(L)})()")
        await ev(pg, "(()=>{const t=L.teams[L.user];for(const id of t.roster){L.players[id].injury=0;L.players[id].injName=null}})()")   # healthy roster: a chart with an injured player falls back to the coach by design
        # ---- game page: rotation chart ----
        gid = await ev(pg, "(()=>{const g=L.games.filter(x=>x.pbp&&x.tl&&(x.home===L.user||x.away===L.user)).slice(-1)[0];return g?g.id:null})()")
        ok(W + 'a played game of yours has a timeline', gid is not None)
        await render(pg, f"ui.view=null;ui.player=null;ui.game={gid};ui.liveMode=null", 700)
        n = await ev(pg, "document.querySelectorAll('.rotchart').length"); bars = await ev(pg, "document.querySelectorAll('.rotchart .rc-track i').length")
        rows = await ev(pg, "document.querySelectorAll('.rotchart .rc-row').length")
        ok(W + 'game page has two rotation charts with bars', n == 2 and bars > 10 and rows >= 16, f'{n} charts, {rows} rows, {bars} bars')
        first = await ev(pg, "(()=>{const r=document.querySelectorAll('.rotchart')[0];const rs=[...r.querySelectorAll('.rc-row')].slice(0,5);return rs.every(x=>x.querySelector('i.st'))})()")
        ok(W + 'first five rows are starters (dark bars)', first)
        mins = await ev(pg, f"(()=>{{const g=L.games.find(x=>x.id=={gid});const r=document.querySelectorAll('.rotchart')[0];const tid=g.away;const si=1;const rows=[...r.querySelectorAll('.rc-row')];const tot=rows.reduce((a,x)=>a+(+x.querySelector('.rc-m').textContent),0);return tot}})()")
        ok(W + 'chart minutes add to about 240', 232 <= mins <= 250, str(mins))
        ok(W + 'game page has no horizontal page scroll', await noscroll())
        # ---- live window: planned subs ----
        await render(pg, f"ui.game={gid};ui.liveMode={gid};ui.lv=null", 700)
        nx = await ev(pg, "[...document.querySelectorAll('.lv-court-tm .lv-next')].map(x=>x.textContent)")
        ok(W + 'starting lineup window shows planned subs for both teams', len(nx) == 2 and all(' in for ' in x for x in nx), json.dumps(nx)[:160])
        ok(W + 'live page has no horizontal page scroll', await noscroll())
        # ---- rotation chart editor ----
        await render(pg, "ui.game=null;ui.liveMode=null;ui.player=null;ui.view='myteam';ui.mtab='roster';ui.teamTab='overview';ui.twin=Object.assign({},ui.twin,{rot:'Rotation chart'});ui.rcDraft=null", 700)
        cols = await ev(pg, "document.querySelectorAll('table.rc-edit thead .rc-bh').length"); btn = await ev(pg, "document.querySelectorAll('.rc-b').length")
        ok(W + 'editor grid: 16 blocks and a toggle per player per block', cols == 16 and btn >= 16 * 10, f'{cols} blocks, {btn} buttons')
        cnt = await ev(pg, "[...document.querySelectorAll('table.rc-edit tbody tr:last-child td')].slice(1,17).map(x=>x.textContent)")
        ok(W + 'coach pattern loaded: every block has 5 on court', all(c == '5' for c in cnt), json.dumps(cnt))
        ok(W + 'save is enabled and Reset disabled for an untouched coach pattern', await ev(pg, "(()=>{const s=document.querySelector('[data-rcact=save]'),r=document.querySelector('[data-rcact=reset]');return !s.disabled&&r.disabled})()"))
        # break a block (add a 6th player) -> cannot save
        await ev(pg, "(()=>{const b=[...document.querySelectorAll('.rc-b:not(.on):not(:disabled)')][0];b.click()})()"); await pg.wait_for_timeout(500)
        st = await ev(pg, "(()=>{const s=document.querySelector('[data-rcact=save]');const m=[...document.querySelectorAll('.msg.no')].map(x=>x.textContent).join(' ');return [s.disabled,m]})()")
        ok(W + 'a block with 6 players blocks saving and says why', st[0] and 'needs 5' in st[1], st[1][:120])
        await pg.click('[data-rcact=revert]'); await pg.wait_for_timeout(500)
        # move the top player out of every block of Q1-Q2 (swap with a bench man) -> role warning
        await ev(pg, "(()=>{const t=L.teams[L.user],d=ui.rcDraft,top=t.roster.map(id=>L.players[id]).sort((a,b)=>b.ovr-a.ovr)[0].id;const bench=t.roster.map(id=>L.players[id]).filter(p=>p.injury<=0&&!d.blocks.some(bl=>bl.includes(p.id)))[0]||t.roster.map(id=>L.players[id]).filter(p=>p.injury<=0&&p.id!==top).sort((a,b)=>a.ovr-b.ovr)[0];for(let i=0;i<16;i++){const bl=d.blocks[i],k=bl.indexOf(top);if(k>=0){bl.splice(k,1);if(!bl.includes(bench.id))bl.push(bench.id);else{const o=t.roster.filter(x=>x!==top&&!bl.includes(x))[0];bl.push(o)}}}d.dirty=true;window.__pbl.render()})()"); await pg.wait_for_timeout(600)
        warn = await ev(pg, "(()=>{const m=[...document.querySelectorAll('.msg')].find(x=>/Minutes vs role targets/.test(x.textContent));return m?m.textContent:null})()")
        ok(W + 'warning when minutes break a role target', bool(warn) and 'target' in warn, (warn or '')[:140])
        # ctor: fix blocks to 5 players -> save
        await pg.click('[data-rcact=revert]'); await pg.wait_for_timeout(400)
        await ev(pg, "(()=>{const b=[...document.querySelectorAll('.rc-b.on:not(:disabled)')][0];b.click();const bn=[...document.querySelectorAll('.rc-b:not(.on):not(:disabled)')].find(x=>x.dataset.rcb.split('|')[1]===b.dataset.rcb.split('|')[1]);bn.click()})()"); await pg.wait_for_timeout(500)
        await pg.click('[data-rcact=save]'); await pg.wait_for_timeout(500)
        saved = await ev(pg, "(()=>{const c=L.teams[L.user].rotChart;return !!c&&c.q.length===4&&c.q.every(q=>q.length===4&&q.every(b=>new Set(b).size===5))})()")
        ok(W + 'Save chart stores a valid t.rotChart (4 x 4 x 5)', saved)
        g = await ev(pg, "(()=>{const t=L.teams[L.user];const a=buildSide(L,t.id,true,false);return a.planSrc})()")
        ok(W + 'the next game would use your chart', g == 'chart', g)
        await pg.click('[data-rcact=reset]'); await pg.wait_for_timeout(500)
        ok(W + 'Reset to coach pattern clears the chart', await ev(pg, "L.teams[L.user].rotChart==null"))
        ok(W + 'editor has no horizontal page scroll', await noscroll())
        # ---- roles + expected minutes ----
        await render(pg, "ui.twin=Object.assign({},ui.twin,{rot:'Rotation & roles'});ui.mtab='roster'", 600)
        cells = await ev(pg, "(()=>{const t=[...document.querySelectorAll('table[data-sid=roster] thead th')].map(x=>x.textContent);const i=t.indexOf('Role');if(i<0)return null;return [...document.querySelectorAll('table[data-sid=roster] tbody tr')].map(r=>r.children[i]&&r.children[i].textContent)})()")
        ok(W + 'roster Role column shows range and expected minutes', bool(cells) and sum(1 for c in cells if 'exp ' in c) >= len(cells) - 4, json.dumps((cells or [])[:3]))
        pid = await ev(pg, "(()=>{const t=L.teams[L.user];return t.roster.map(id=>L.players[id]).sort((a,b)=>b.ovr-a.ovr)[2].id})()")
        await render(pg, f"ui.view=null;ui.player={pid}", 600)
        box = await ev(pg, "(()=>{const k=[...document.querySelectorAll('.sumcard .k')].find(x=>x.textContent==='Role');return k?k.nextElementSibling.textContent:null})()")
        ok(W + 'player page Role box shows range and plan minutes', bool(box) and 'plan' in box and 'min' in box, box or '')
        ok(W + 'player page has no horizontal page scroll', await noscroll())
        # ---- offer popup: role minutes + mid-level ----
        fa = await ev(pg, "(()=>{const f=L.fa.map(id=>L.players[id]).filter(p=>p&&!p.prospect&&p.ovr>=55).sort((a,b)=>b.ovr-a.ovr)[0];return f?f.id:null})()")
        await ev(pg, "(()=>{const t=L.teams[L.user];t.dead=[{pid:-9,name:'x',amt:Math.max(0,TAXLINE+2-payroll(L,t)),exp:capSeason(L)}]})()")
        await render(pg, f"L.phase='regular';ui.player={fa};ui.neg={{}}", 600)
        await pg.click('#negbar [data-nego]'); await pg.wait_for_timeout(700)
        ml = await ev(pg, "(()=>{const r=[...document.querySelectorAll('#modal .muted')].find(x=>/Coach would likely use him as/.test(x.textContent));return r?r.textContent:null})()")
        ok(W + 'offer form shows the likely role and minutes', bool(ml) and 'min a game' in ml, (ml or '')[:100])
        await ev(pg, "(()=>{const s=document.querySelector('#neg-role');s.value='starter';s.dispatchEvent(new Event('change',{bubbles:true}))})()"); await pg.wait_for_timeout(700)
        ml2 = await ev(pg, "(()=>{const r=[...document.querySelectorAll('#modal .muted')].find(x=>/You promise/.test(x.textContent));return r?r.textContent:null})()")
        ok(W + 'promised role shows its minutes range', bool(ml2) and '26-32' in ml2, (ml2 or '')[:100])
        mb = await ev(pg, "(()=>{const m=document.querySelector('#mlebox');return m?m.textContent:null})()")
        ok(W + 'offer popup has the mid-level box (amount left, used or not)', bool(mb) and 'Mid-level exception' in mb and ('left' in mb), (mb or '')[:140])
        ok(W + 'cannot use it here (payroll over the tax line): checkbox disabled with a reason', await ev(pg, "(()=>{const c=document.querySelector('#neg-mle');return !!c&&c.disabled})()") and "Can't use it on this offer" in (mb or ''), (mb or '')[:160])
        ok(W + 'offer popup has no horizontal page scroll', await noscroll())
        await ev(pg, "document.querySelector('#modal').innerHTML='';ui.negOpen=null")
        # make payroll just under the cap so the MLE can be used
        await ev(pg, "(()=>{const t=L.teams[L.user];t.mleUsed=null;t.dead=[];const ps=t.roster.map(id=>L.players[id]).sort((a,b)=>b.contract.amt-a.contract.amt);for(const p of ps){if(payroll(L,t)<=CAP-3)break;p.contract.amt=MINSAL}t.dead=[{pid:-9,name:'x',amt:Math.max(0,CAP-2-payroll(L,t)),exp:capSeason(L)}];ui.neg={};window.__pbl.render()})()")
        pay = await ev(pg, "payroll(L,L.teams[L.user])")
        await render(pg, f"ui.player={fa}", 500)
        await pg.click('#negbar [data-nego]'); await pg.wait_for_timeout(700)
        await ev(pg, f"(()=>{{const i=document.querySelector('#neg-amt');i.value='{ 8 }';i.dispatchEvent(new Event('change',{{bubbles:true}}))}})()"); await pg.wait_for_timeout(700)
        el = await ev(pg, "(()=>{const c=document.querySelector('#neg-mle');return c?[c.disabled,c.checked]:null})()")
        ok(W + 'offer that crosses the cap under the tax line: Use mid-level enabled and ticked', el == [False, True], f'{el} pay {pay}')
        await ev(pg, "(()=>{const c=document.querySelector('#neg-mle');c.click()})()"); await pg.wait_for_timeout(700)
        el2 = await ev(pg, f"(()=>{{const c=document.querySelector('#neg-mle');return [c.checked,(ui.neg[{fa}]||{{}}).mle]}})()")
        ok(W + 'unticking is remembered in the offer state (mle:false)', el2 == [False, False], json.dumps(el2))
        await ev(pg, f"(()=>{{const i=document.querySelector('#neg-amt');i.value='60';i.dispatchEvent(new Event('change',{{bubbles:true}}))}})()"); await pg.wait_for_timeout(700)
        wt = await ev(pg, "(()=>{const m=document.querySelector('#mlebox');return m?m.textContent:null})()")
        ok(W + 'crossing the tax line / hard cap gives a warning', bool(wt) and ('tax line' in wt or 'hard cap' in wt) , (wt or '')[:200])
        await ev(pg, "document.querySelector('#modal').innerHTML='';ui.negOpen=null")
        # cap box
        await render(pg, "ui.player=null;ui.view='myteam';ui.mtab='cap'", 600)
        cb = await ev(pg, "(()=>{const x=[...document.querySelectorAll('.panel p.muted')].find(e=>/mid-level exception/.test(e.textContent));return x?x.textContent:null})()")
        ok(W + 'cap box shows the mid-level amount left and used or not', bool(cb) and 'left' in cb and 'this season' in cb, (cb or '')[:200])
        # ---- coaching targets ----
        await render(pg, "ui.view='fa';ui.player=null;L.phase='off-fa'", 800)
        th = await ev(pg, "[...document.querySelectorAll('th')].some(x=>x.textContent==='Cap fit')")
        notes = await ev(pg, "[...document.querySelectorAll('td.capfit')].map(x=>x.textContent)")
        ok(W + 'coach targets (free agents) have a Cap fit column with a note on every row', th and len(notes) > 0 and all(len(n) > 8 for n in notes), json.dumps(notes[:2]))
        ok(W + 'free agents page has no horizontal page scroll', await noscroll())
        # ---- draft tab: team picks ----
        await render(pg, "ui.view='draft';ui.dpage='picks';ui.dpTeam=null;L.phase='regular'", 800)
        r1 = await ev(pg, "document.querySelectorAll('table.pk-table tbody tr:not(.pkhead)').length")
        ok(W + 'Team picks lists your own picks by draft year', r1 >= 4, str(r1))
        await ev(pg, "(()=>{const o=L.teams.find(t=>t.id!==L.user),k=L.picks.find(k=>k.owner===o.id&&k.rd===1&&k.s>=L.season);k.owner=L.user;k.prot=5;k.roll=0;ui.tmpPick=[o.abbr,k.id]})()")
        await render(pg, "ui.dpage='picks'", 600)
        tx = await ev(pg, "document.querySelector('table.pk-table').textContent")
        ok(W + 'a traded pick shows "via" the original team and its protection', ('via' in tx) and 'protected' in tx, tx[:40])
        other = await ev(pg, "L.teams.find(t=>t.id!==L.user).id")
        await pg.select_option('#dp-team', str(other)); await pg.wait_for_timeout(600)
        ok(W + 'team picker switches teams', await ev(pg, f"ui.dpTeam=={other}") and await ev(pg, "document.querySelectorAll('table.pk-table tbody tr:not(.pkhead)').length")>0)
        ok(W + 'team picks page has no horizontal page scroll', await noscroll())
        ok(W + 'no page errors', not errs, '; '.join(errs[:2]))
        await b.close()
async def main():
    await run(1300); await run(390)
    print('\n'.join(res)); print(f"\nbrowser-v32d: {sum(r.startswith('PASS') for r in res)}/{len(res)}")
asyncio.run(main())
