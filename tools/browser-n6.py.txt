# v27 N6 browser test. Serve the artifact folder, then: python3 browser-n6.py PORT
import asyncio, sys, re
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8779'
res = []
def ok(n, c, x=''): res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else ''))
CLR = "document.querySelector('#modal').innerHTML=''"
async def ev(pg, js): return await pg.evaluate(js)
async def go(pg, h):
    await ev(pg, CLR); await ev(pg, "location.hash='#standings'"); await pg.wait_for_timeout(200)
    await ev(pg, f"location.hash='#{h}'"); await pg.wait_for_timeout(700)
async def txt(pg): return await ev(pg, "document.querySelector('#main').innerText")
async def tc(pg): return await ev(pg, "document.querySelector('#main').textContent")
def clean(t): return '{' not in t and 'undefined' not in t and 'NaN' not in t
async def main():
  try:
    async with async_playwright() as p:
        b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
        ctx = await b.new_context(viewport={'width': 1300, 'height': 900}); pg = await ctx.new_page()
        errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
        await pg.click('[data-setmode="team"]'); await pg.click('[data-setnba="1"]'); await pg.wait_for_timeout(500)
        await pg.click('[data-setmode="team"]')
        st = await txt(pg)
        ok('setup shows East / West conferences', 'east conference' in st.lower() and 'west conference' in st.lower() and 'north conference' not in st.lower())
        await pg.click('.teampick:has-text("Thunder")'); await pg.click('#create'); await pg.wait_for_timeout(5000); await ev(pg, CLR)
        ok('NBA league created (Franchise, OKC)', await ev(pg, "(()=>{const L=__pbl.L;return !!L.nba&&L.teams.length===30&&L.teams[L.user].abbr==='OKC'})()"))
        ok('title / brand NBA 2026-27', 'NBA 2026-27' in (await ev(pg, "document.title")) and 'NBA 2026-27' in (await ev(pg, "document.querySelector('#brandname').textContent")))
        await go(pg, 'standings'); t = await txt(pg)
        ok('standings East / West + 6 divisions', 'east conference' in t.lower() and 'west conference' in t.lower() and all(d in t for d in ['Atlantic','Central','Southeast','Northwest','Pacific','Southwest']) and not re.search(r'\b(North|South)\b', t, re.I) and clean(await tc(pg)))
        await go(pg, 'teams'); t = await txt(pg); ok('teams page no market / region text', 'market' not in t.lower() and clean(await tc(pg)))
        await ev(pg, "(()=>{const L=__pbl.L;const p=Object.values(L.players).filter(x=>x.team===L.user);window.__pbl.openPlayer(p.find(x=>/Gilgeous/.test(x.name)||/Gilgeous/.test(x.last)).id)})()"); await pg.wait_for_timeout(500)
        t = await txt(pg); ok('OKC roster has Shai Gilgeous-Alexander (player page opens)', 'Gilgeous' in t and clean(await tc(pg)))
        ok('player page shows contract years', 'through' in t, re.findall(r'.{20}through.{14}', t)[:1])
        await ev(pg, "__pbl.ui.player=null;__pbl.render()")
        await go(pg, 'roster'); t = await txt(pg); ok('roster page lists SGA', 'Gilgeous' in t)
        # sim regular season
        await ev(pg, "(()=>{const L=__pbl.L;let g=0;while(L.phase==='regular'&&g++<400)simDay(L);__pbl.render()})()")
        ok('regular season ended', await ev(pg, "__pbl.L.teams.every(t=>t.w+t.l===82)"))
        await go(pg, 'standings'); ok('end-of-season standings clean', clean(await tc(pg)))
        await ev(pg, "(()=>{const L=__pbl.L;processPending(L);startPlayoffs(L);__pbl.render()})()")
        ph = await ev(pg, "__pbl.L.phase"); ok('play-in / playoffs start', ph in ('playin','playoffs'), ph)
        await ev(pg, "(()=>{const L=__pbl.L;let g=0;while((L.phase==='playin'||L.phase==='playoffs')&&g++<600)simPlayoffDay(L);__pbl.render()})()")
        ok('playoffs finished with a champion', await ev(pg, "(()=>{const L=__pbl.L;return L.phase==='season-done'&&L.history.some(h=>h.s===L.season&&h.champion!=null)||L.phase==='season-done'})()"), await ev(pg, "__pbl.L.phase"))
        await ev(pg, CLR); await go(pg, 'standings'); t = await txt(pg)
        ok('bracket page East / West, no North / South', ('east' in t.lower()) and not re.search(r'\b(North|South)\b', t, re.I) and clean(await tc(pg)), ('east' in t.lower(), clean(await tc(pg)), re.findall(r'[^\n]{0,30}\b(?:North|South)\b[^\n]{0,30}', t, re.I)[:4]))
        # offseason
        await ev(pg, "(()=>{const L=__pbl.L;startOffseason(L);runLottery(L);startDraft(L);__pbl.render()})()")
        await go(pg, 'draft'); ok('draft page clean', clean(await tc(pg)))
        await ev(pg, "(()=>{const L=__pbl.L;let g=0;while(L.phase==='off-draft'&&g++<200)aiDraftPick(L);startFA(L);g=0;while(L.phase==='off-fa'&&g++<40)faDay(L);processPending(L);startNewSeason(L);__pbl.render()})()")
        ok('new season started (2027-28)', await ev(pg, "__pbl.L.phase==='regular'") and '2027-28' in await ev(pg, "document.querySelector('#when').textContent"), await ev(pg, "document.querySelector('#when').textContent"))
        # save / load
        await ev(pg, "__pbl.save()"); await pg.wait_for_timeout(1500)
        await pg.reload(); await pg.wait_for_timeout(3500); await ev(pg, CLR)
        ok('save/load keeps NBA', await ev(pg, "(()=>{const L=__pbl.L;return !!L&&!!L.nba&&L.teams.length===30&&L.teams[L.user].abbr==='OKC'})()"))
        await go(pg, 'standings'); t = await txt(pg); ok('reloaded standings East / West', 'east conference' in t.lower() and 'west conference' in t.lower() and clean(await tc(pg)), t[:60])
        ok('no page errors', not errs, '; '.join(errs)[:300])
        await b.close()
  finally:
    print('\n'.join(res)); print(sum(r.startswith('PASS') for r in res), '/', len(res))
asyncio.run(main())
