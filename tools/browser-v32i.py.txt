# S6b (v32i) browser check: team history (franchise leaders, retired numbers, seasons), League records, Hall of Fame, Seasons best record.
# Based on the v32f-a check (desktop 1300 and 390px): History > Seasons / Championships / Awards (one list, no Real NBA tab), a team page (Overview + History),
# a real player's page (career table, trophy case, rating chart note), Dynasties & eras. Needs a saved league with 1+ sim seasons:
#   LOAD: SAVE=/tmp/nba-s1.json node tools/t32f-a.js 1      (writes the league json; ~90 s)
# Rename to .py, copy the json into the game folder as tools/_s1.json, serve the folder (python3 -m http.server PORT), then:
#   python3 browser-v32f-a.py PORT [SHOTS_DIR]
import asyncio, sys, json, os
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8793'
SHOTS = sys.argv[2] if len(sys.argv) > 2 else None
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui;return eval(" + json.dumps(js) + ")})()")
async def render(pg, js='0', wait=600):
    await ev(pg, "(()=>{" + js + ";window.__pbl.render()})()"); await pg.wait_for_timeout(wait)
async def run(width):
    async with async_playwright() as p:
        try: b = await p.chromium.launch()
        except Exception: b = await p.chromium.launch(executable_path='/usr/bin/google-chrome')
        pg = await (await b.new_context(viewport={'width': width, 'height': 900})).new_page(); errs = []
        pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2000)
        # put the saved league into the app's own storage, then reload so the app loads it like any save
        await pg.evaluate("""async () => { const txt = await (await fetch('tools/_s1.json')).text();
          const db = await new Promise((res, rej) => { const r = indexedDB.open('pbl-league', 1); r.onupgradeneeded = () => r.result.createObjectStore('s'); r.onsuccess = () => res(r.result); r.onerror = () => rej(r.error); });
          const put = (k, v) => new Promise((res, rej) => { const tx = db.transaction('s', 'readwrite'); tx.objectStore('s').put(v, k); tx.oncomplete = res; tx.onerror = () => rej(tx.error); });
          await put('league:slot-t32', txt); await put('slots', JSON.stringify([{ id: 'slot-t32', name: 't32', mode: 'commish', season: 2, phase: 'off-draft', updated: Date.now() }])); await put('current', 'slot-t32'); db.close(); }""")
        await pg.reload(); await pg.wait_for_timeout(4500)
        W = f'{width}px: '
        noscroll = lambda: ev(pg, "document.documentElement.scrollWidth<=window.innerWidth+2")
        await ev(pg, "document.querySelector('#modal') && (document.querySelector('#modal').innerHTML='')")
        nh = await ev(pg, "L.history.length"); ok(W + 'saved NBA league loaded with a sim season', nh >= 1 and await ev(pg, "!!L.nba"), str(nh))
        main = lambda: ev(pg, "document.querySelector('#main').innerText")
        shot = lambda n: pg.screenshot(path=f'{SHOTS}/{n}-{width}.png') if SHOTS else asyncio.sleep(0)
        for view, js, need in [
          ('team history', "history.replaceState(null,'','#teams');ui.view='teams';ui.team=L.teams.find(t=>(t.nbaAb||t.abbr)==='CHI').id;ui.teamTab='history';ui.player=null;ui.game=null", ['Franchise leaders', 'Retired numbers', 'Franchise seasons', 'Michael Jordan']),
          ('team history playoffs', "ui.hxTy='PO'", ['Franchise leaders', 'Michael Jordan']),
          ('league records', "history.replaceState(null,'','#history');ui.view='history';ui.histTab='records';ui.hs=null;ui.team=null", ['Wilt Chamberlain', 'Box plus-minus']),
          ('awards + HOF', "ui.view='history';ui.histTab='awards';ui.hs=null", ['Hall of Fame']),
          ('seasons', "ui.view='history';ui.histTab='seasons';ui.hs=null", ['72-10'])]:
            await render(pg, js, 900); t = await main()
            ok(W + view + ': shows ' + ', '.join(need), all(x.lower() in t.lower() for x in need), ','.join(x for x in need if x.lower() not in t.lower()))
            ok(W + view + ': no real-vs-game text', 'Real NBA' not in t and 'real NBA' not in t and '(real' not in t)
            ok(W + view + ': no horizontal page scroll', await noscroll())
            await shot(view.replace(' ', '-'))
        ok(W + 'no page errors', not errs, '; '.join(errs[:3]))
        await b.close()
async def main():
    for w in (1300, 390): await run(w)
    print('\n'.join(res)); print(f"{sum(r.startswith('PASS') for r in res)}/{len(res)} pass")
asyncio.run(main())
