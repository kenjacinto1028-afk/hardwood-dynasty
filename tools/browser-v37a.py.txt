# short tag chips (no duplicate cornerstone chip), one ⋯ menu per row (desktop click + 390px tap), same actions on the player page,
# then v37b: History > League records sub tabs and Pipeline > Overview. Base: tools/browser-v30t.py.txt + browser-v36d.py.txt loader.
# gunzip -c tools/snap/nba-team-s10.json.gz > tools/_t1.json; rename to .py, serve the game folder (python3 -m http.server PORT), then:
#   python3 browser-v37a.py PORT [PAGE] [SAVE]
import asyncio, sys, json
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8801'
PAGE = sys.argv[2] if len(sys.argv) > 2 else 'index.html'
SAVE = sys.argv[3] if len(sys.argv) > 3 else 'tools/_t1.json'
res = []
def ok(n, c, x=''):
    res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else '')); print(res[-1], file=sys.stderr, flush=True)
async def ev(pg, js): return await pg.evaluate("(()=>{const L=window.__pbl.L,ui=window.__pbl.ui;return eval(" + json.dumps(js) + ")})()")
async def show(pg, js, wait=700):
    await ev(pg, "document.querySelector('#modal').innerHTML='';ui.player=null;ui.game=null;ui.coach=null;ui.team=null;" + js + ";location.hash='#'+ui.view;window.__pbl.render()"); await pg.wait_for_timeout(wait)
# every table's natural width (width:auto) vs its scroll box
NAT = "[...document.querySelectorAll('#main table')].map(t=>{const w=t.closest('.tblwrap');const cw=w?w.clientWidth:9999;const o=t.style.width;t.style.width='auto';const n=t.offsetWidth;t.style.width=o;return {n,cw,rows:t.rows.length}}).filter(x=>x.n>x.cw+1)"
DOCW = "document.documentElement.scrollWidth<=innerWidth+1"
PAGES = {
  'roster': "ui.view='myteam';ui.teamTab='overview';ui.mtab='roster'",
  'draft board': "ui.view='draft';ui.dpage='cls';ui.dtab='board'",
  'trade center': "ui.view='trades';ui.trTab='center'",
}
async def load(p, width, touch=False):
    b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
    ctx = await b.new_context(viewport={'width': width, 'height': 900}, has_touch=touch, is_mobile=touch)
    pg = await ctx.new_page(); errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
    await pg.goto(f'http://localhost:{PORT}/{PAGE}'); await pg.wait_for_timeout(2000)
    await pg.evaluate("""async (f) => { const txt = await (await fetch(f)).text();
      const db = await new Promise((res, rej) => { const r = indexedDB.open('pbl-league', 1); r.onupgradeneeded = () => r.result.createObjectStore('s'); r.onsuccess = () => res(r.result); r.onerror = () => rej(r.error); });
      const put = (k, v) => new Promise((res, rej) => { const tx = db.transaction('s', 'readwrite'); tx.objectStore('s').put(v, k); tx.oncomplete = res; tx.onerror = () => rej(tx.error); });
      await put('league:slot-t36', txt); await put('slots', JSON.stringify([{ id: 'slot-t36', name: 't36', mode: 'team', season: 11, phase: 'regular', updated: Date.now() }])); await put('current', 'slot-t36'); }""", '/' + SAVE)
    await pg.reload(); await pg.wait_for_timeout(6000)
    return b, pg, errs
def rmOpen(pg, touch): return pg.tap if touch else pg.click
async def menu_state(pg):
    return await ev(pg, "(()=>{const m=document.querySelector('#rowmenu');if(!m||m.hidden)return null;const r=m.getBoundingClientRect();return {items:[...m.querySelectorAll('.rmi')].map(x=>x.textContent.trim()),inView:r.left>=0&&r.top>=0&&r.right<=innerWidth+0.5&&r.bottom<=innerHeight+0.5,sheet:m.classList.contains('sheet'),w:Math.round(r.width)}})()")

async def load_new(p, width, touch=False):
    b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
    ctx = await b.new_context(viewport={'width': width, 'height': 900}, has_touch=touch, is_mobile=touch)
    pg = await ctx.new_page(); errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
    await pg.goto(f'http://localhost:{PORT}/{PAGE}'); await pg.wait_for_timeout(2500)
    await pg.click('[data-setnba="1"]'); await pg.wait_for_timeout(300)
    await pg.click('[data-setmode="commish"]')
    await pg.click('#create'); await pg.wait_for_timeout(6000)
    await ev(pg, "document.querySelector('#modal').innerHTML=''")
    return b, pg, errs
