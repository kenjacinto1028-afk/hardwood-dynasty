# v26.6b Sonnet checks (S1-S4 labels / dates). Serve the artifact folder, then: python3 browser-v266b.py PORT
import asyncio, sys, re
from playwright.async_api import async_playwright
PORT = sys.argv[1] if len(sys.argv) > 1 else '8779'
res = []
def ok(n, c, x=''): res.append(('PASS ' if c else 'FAIL ') + n + (f'  [{x}]' if x else ''))
CLR = "document.querySelector('#modal').innerHTML=''"
async def go(pg, h):
    await pg.evaluate(CLR); await pg.evaluate("location.hash='#standings'"); await pg.wait_for_timeout(200)
    await pg.evaluate(f"location.hash='#{h}'"); await pg.wait_for_timeout(700)
async def txt(pg): return (await pg.evaluate("document.querySelector('#main').innerText"))
async def tc(pg): return await pg.evaluate("document.querySelector('#main').textContent")
def clean(t): return '{' not in t and 'undefined' not in t and 'NaN' not in t
async def ev(pg, js): return await pg.evaluate(js)
async def when(pg): return await pg.evaluate("document.querySelector('#when').textContent")
async def main():
  try:
    async with async_playwright() as p:
        b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
        pg = await (await b.new_context(viewport={'width': 1300, 'height': 900})).new_page()
        errs = []; pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto(f'http://localhost:{PORT}/index.html'); await pg.wait_for_timeout(2500)
        await pg.click('[data-setmode="team"]'); await pg.click('#create'); await pg.wait_for_timeout(4500); await ev(pg, CLR)
        ok('new league has y0 2025', await ev(pg, "window.__pbl.L.y0===2025"))
        # S1 / S4 header: season name + calendar date
        for i in range(3): await ev(pg, "(()=>{const L=window.__pbl.L;if(L.phase!=='regular')return;window.__pbl.render()})()")
        await ev(pg, "(()=>{const L=window.__pbl.L;for(let i=0;i<25&&L.phase!=='regular';i++){}window.__pbl.render()})()")
        w = await when(pg)
        ok('S1 header shows 2026-27', '2026-27' in w, w)
        ok('S4 header shows Oct 20, 2026 and Day 1', 'Oct 20, 2026' in w and 'Day 1' in w, w)
        await ev(pg, "(()=>{const L=window.__pbl.L;for(let i=0;i<45;i++)simDay(L);window.__pbl.render()})()")
        w = await when(pg)
        ok('S4 header date moves on', re.search(r'(Nov|Dec) \d+, 2026', w) is not None and 'Day 46' in w, w)
        await go(pg, 'schedule'); t = await txt(pg)
        ok('S4 schedule shows dates', re.search(r'(Oct|Nov|Dec|Jan) \d+, 202[67]', t) is not None and clean(await tc(pg)))
        await go(pg, 'standings'); t = await txt(pg); ok('S1 standings clean', clean(await tc(pg)))
        # player page season rows
        await ev(pg, "(()=>{const L=window.__pbl.L;const p=rosterOf(L,L.teams[L.user])[0];window.__pbl.ui.player=p.id;window.__pbl.render()})()"); await pg.wait_for_timeout(400); t = await txt(pg)
        ok('S1 player page shows 2026-27', '2026-27' in t and clean(await tc(pg)))
        ok('S3 contract says through 20xx-xx', re.search(r'through 20\d\d-\d\d', t) is not None, re.findall(r'.{20}through.{14}', t)[:1])
        await ev(pg, "window.__pbl.ui.player=null;window.__pbl.render()")
        # S2 draft
        await go(pg, 'draft'); t = await txt(pg)
        ok('S2 draft page shows 2027', '2027' in t and clean(await tc(pg)), t[:80].replace('\n', ' '))
        ok('S2 draft page has no 2026 draft label', not re.search(r'2026 (draft|class)', t))
        await ev(pg, "(()=>{const L=window.__pbl.L;const p=Object.values(L.players).find(x=>x.prospect&&!x.retired);window.__pbl.ui.player=p.id;window.__pbl.render()})()"); await pg.wait_for_timeout(400); t = await txt(pg)
        ok('S2 prospect page: 2027 draft prospect / class', '2027 draft prospect' in t.lower() and '2027 class' in t.lower(), re.findall(r'.{10}2027.{15}', t)[:2])
        await ev(pg, "window.__pbl.ui.player=null;window.__pbl.render()")
        await go(pg, 'history'); ok('S1 history page clean', clean(await tc(pg)))
        # S4 deadline / All-Star text
        await go(pg, 'trades'); t = await txt(pg)
        ok('S4 trade page shows the deadline date', 'Feb 18, 2027' in t or re.search(r'on (Jan|Feb|Mar) \d+, 2027', t) is not None, re.findall(r'.{30}deadline.{40}', t)[:1])
        # old save: no y0
        await ev(pg, "(()=>{const L=window.__pbl.L;delete L.y0;setSeasonLen(L);window.__pbl.render()})()"); await pg.wait_for_timeout(300)
        w = await when(pg)
        ok('old save header shows 2000, no date', '2000' in w and '2026' not in w and 'Oct 20' not in w and 'Day 46' in w, w)
        await go(pg, 'schedule'); t = await txt(pg)
        ok('old save schedule keeps Day N', 'Day ' in t and not re.search(r'\w{3} \d+, 20', t) and clean(await tc(pg)))
        await ev(pg, "(()=>{const L=window.__pbl.L;const p=rosterOf(L,L.teams[L.user])[0];window.__pbl.ui.player=p.id;window.__pbl.render()})()"); await pg.wait_for_timeout(400); t = await txt(pg)
        ok('old save player page shows 2000, no 2026-27', '2000' in t and '2026-27' not in t and clean(await tc(pg)))
        await ev(pg, "window.__pbl.ui.player=null;window.__pbl.render()")
        await go(pg, 'draft'); t = await txt(pg); ok('old save draft page clean', clean(await tc(pg)) and '2027' not in t)
        ok('no page errors', not errs, '; '.join(errs)[:300])
        await b.close()
  finally:
    print('\n'.join(res)); print(sum(r.startswith('PASS') for r in res), '/', len(res))
asyncio.run(main())
